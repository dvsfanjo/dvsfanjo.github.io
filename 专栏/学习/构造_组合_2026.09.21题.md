## arc103b Robot Arms



一开始想的就是倍增构造，但是 $m\le40$ 卡的太紧遂放弃。事实上倍增构造不需要横纵坐标都准备 $2^0 \to 2^{30}$ 那么多，画一下图就能发现只要一套倍增的机械臂就能覆盖所有横纵坐标之和为奇数的点（若是偶数可以通过加一个长为 $1$ 的机械臂解决）。观察题目样例，发现有解的样例中 $|x_i|+|y_i|$ 相等，进而想到 $|x_i+y_i|$ 的奇偶性相等为判定。最后就是暴力从大到小使用机械臂，将差值更大的坐标用当前机械臂向 $0$ 趋近，正确性不显然，大致想到是通过一些神奇抵消得到的，反正想到这就差不多直接写了。



:::info[代码]

```cpp

#include <bits/stdc++.h>

#define int long long

#define fi first

#define se second

using namespace std;

const int maxn = 1e5 + 5;

int n, m, x[maxn], y[maxn], tmp, len[maxn]; 

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> x[i] >> y[i];

		int lst = tmp;

		tmp = abs(x[i] + y[i]) & 1; 

		if (i > 1 && tmp != lst) return cout << -1, 0;

	}

	cout << 32 - tmp << '\n' << 1 << ' ';

	len[++m] = 1;

	for (int i = tmp; i <= 30; i++) cout << (len[++m] = (1 << i)) << ' '; 

	cout << '\n';

	for (int i = 1; i <= n; i++) {

		vector <char> s(m + 1);

		int nx = 0, ny = 0;

		for (int j = m; j; j--) {

			int dx = x[i] - nx, dy = y[i] - ny;

			if (abs(dx) > abs(dy)) {

				if (dx > 0) nx += len[j], s[j] = 'R';

				else nx -= len[j], s[j] = 'L';

			} else {

				if (dy > 0) ny += len[j], s[j] = 'U';

				else ny -= len[j], s[j] = 'D';

			}

		}

		for (int j = 1; j <= m; j++) cout << s[j];

		cout << '\n';

	}

	return 0;

}
```

## arc103c Tr/ee



无解判断瞪眼法就能看出来，主要是构造。首先不难看出对于一条链和 $s$ 的对应关系是显然的，若 $s_i=1$ 则在链上新加点，否则对子树大小为 $i$ 的结点下再挂一个结点。



:::info[代码]

```cpp

#include <bits/stdc++.h>

#define int long long

#define fi first

#define se second

using namespace std;

const int maxn = 1e5 + 5;

int n, now = 1, cnt = 1;

string s;

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> s;

	n = s.size();

	s = " " + s;

	if (s[1] == '0' || s[n] == '1') return cout << -1, 0;

	for (int i = 1; i < n; i++) {

		if (s[i] != s[n - i]) return cout << -1, 0;

	}

	for (int i = 1; i <= n - i; i++) {

		if (s[i] == '1') {

			cout << now << ' ' << cnt + 1 << '\n';

			now = cnt = cnt + 1;

		} else {

			cout << now << ' ' << ++cnt << '\n';

		}

	}

	while (cnt < n) cout << now << ' ' << ++cnt << '\n';

	return 0;

}
```

## arc102c Stop. Otherwise...



对于每一个 $i$ 看怎么算答案，计算出受束缚的点对数量，然后讨论。受束缚的点对是形如 $(x,i-x)$ 或 $(x,x)$ 的，两种情况分一下边界条件就能算出总共有 $s$ 对受束缚的数，剩下 $t=k-s$ 则随便取用。枚举一个选取 $j$ 个受束缚的数，然后统计答案即可，很简单的组合计数。



:::info[代码]

```cpp

#include <bits/stdc++.h>

#define int long long

#define fi first

#define se second

using namespace std;

const int maxn = 4005, p = 998244353;

int n, k, fac[maxn];

int ksm(int a, int b) {

	int res = 1;

	while (b) {

		if (b & 1) res = res * a % p;

		a = a * a % p, b >>= 1;

	}

	return res;

}

int C(int n, int m) {

	if (m < 0 || m > n) return 0;

	return fac[n] * ksm(fac[m], p - 2) % p * ksm(fac[n - m], p - 2) % p;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> k >> n;

	fac[0] = 1;

	for (int i = 1; i <= 4000; i++) {

		fac[i] = fac[i - 1] * i % p;

	}

	for (int i = 2; i <= (k << 1); i++) {

		int l = max(1ll, i - k), r = (i - 1) >> 1;

		int a = l <= r ? r - l + 1 : 0, b = 0;

		if ((i & 1 ^ 1) && 1 <= i / 2 && i / 2 <= k) b = 1;

		int s = a + b, t = k - s, ans = 0;

		for (int j = 0; j <= min(n, s); j++) {

			(ans += C(s, j) % p * C(n + t - j - 1, n - j) % p) %= p;

		}

		cout << ans << '\n';

	}

	return 0;

}

```
