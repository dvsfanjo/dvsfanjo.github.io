## CF1553F



求每个 $ans_k=\sum_{1\le i,j\le k} a_i\mod a_j$ ，且 $a_i$ 不重复，值域与 $n$ 同阶。



首先单独求答案一看就不可做，于是递推的做，得到：



$$ans_i=ans_{i-1}+\sum_{j=1}^{i-1}(a_j\mod a_i)+\sum_{j=1}^{i-1}(a_i\mod a_j)$$



第一个枚举区间 $[ka_i,(k+1)a_i)$ 然后用线段树计数得到和与个数，时间复杂度 $O(\frac n{a_i}\log n)$ ，因为 $a_i$ 不重复的优秀性质，整体答案因为调和级数变为 $O(n\log^2n)$ 。



第二个转化一下变成 $(i-1)a_i-\sum_{i<j}\lfloor\frac{a_i}{a_j}\rfloor a_j$ ，给所有的 $b_{ka_i}$ 加上 $a_i$ 然后求和号后面的部分就是 $\sum_{j=1}^{a_i}b_j$ 。



总时间复杂度 $O(n\log^2n)$ 。



:::info[代码]

```cpp

#include <bits/stdc++.h>

#define int long long

#define fi first

#define se second

using namespace std;

const int maxn = 3e5 + 5;

int n, a[maxn], ans[maxn]; 

struct BIT {

	int t[maxn];

	BIT() { memset(t, 0, sizeof t); }

	void update(int x, int y) { x++; for (; x <= 300001; x += x & -x) t[x] += y; }

	int query(int l, int r, int sum = 0) {

		l++, r++;

		for (int x = l - 1; x; x -= x & -x) sum -= t[x];

		for (int x = r; x; x -= x & -x) sum += t[x];

		return sum;

	}

} t1, t2, t3;

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

	}

	for (int i = 1; i <= n; i++) {

		ans[i] = ans[i - 1];

		for (int k = 0; k * a[i] <= 300000; k++) {

			int l = k * a[i], r = min(300000ll, (k + 1) * a[i] - 1);

			int sum = t1.query(l, r), cnt = t2.query(l, r);

			ans[i] += sum - cnt * k * a[i];

		}

		ans[i] += (i - 1) * a[i] - t3.query(1, a[i]);

		for (int k = 1; k * a[i] <= 300000; k++) {

			t3.update(k * a[i], a[i]);

		}

		t1.update(a[i], a[i]), t2.update(a[i], 1);

		cout << ans[i] << ' ';

	}

	return 0;

}
```

## CF1665E



多次询问，求 $\min_{l\le i<j\le r}a_i|a_j$ 。



原题思路很巧妙，但是不好想到，所以一步一步想。首先发现区间中可选的 $(i,j)$ 一定前 $k$ 小中，因为要使最高位少出现。



现在最重要的就是 $k$ 了，但是很难直接发现规律，于是打表发现 $n$ 个数的答案一定在前 $\log n+1$ 小的数中。其实知道 $k$ 比较小时就可以略过证明直接做了，一般来说为了保险直接维护前 $40$ 小即可。



可是证明也是必要的，至少不在考场上遇到这种题时要试着证明。即证明在 $2^k$ 个数中，答案一定在 $k+1$ 内，而 $k=1$ 时一定成立。然后归纳法证明，对于 $k>1$ ，若最高位上都是 $1$ 则这一位必选，答案在 $k-1$ 中；若最高位上有至少 $2$ 个 $0$ 则这一位就可以定为 $0$ 了，答案也在 $k-1$ 中；最后若最高位只有 $1$ 个 $0$ 则最高位固定为 $1$ 答案在 $k-1$ 中。



神奇吧，因为第 $k$ 位一定是固定的，所以答案就会回到 $k-1$ 中，然后归纳法就证毕了。



:::info[代码]

```cpp

#include <bits/stdc++.h>

#define int long long

#define fi first

#define se second

using namespace std;

const int maxn = 1e5 + 5;

int T, n, q, a[maxn];

struct node {

	int tot, minn[32];

	node() { tot = 0, memset(minn, 0, sizeof minn); }

	friend node operator + (node a, node b) {

		vector <int> vec; node c;

		for (int i = 1; i <= a.tot; i++) vec.push_back(a.minn[i]);

		for (int i = 1; i <= b.tot; i++) vec.push_back(b.minn[i]);

		sort(vec.begin(), vec.end());

		for (int x : vec) {

			c.minn[++c.tot] = x;

			if (c.tot == 31) break;

		}

		return c;

	}

} t[maxn << 2];

#define ls (i << 1)

#define rs (ls | 1)

#define mid (l + r >> 1)

void build(int l, int r, int i) {

	if (l == r) {

		t[i].tot = 0, memset(t[i].minn, 0, sizeof t[i].minn); 

		t[i].minn[++t[i].tot] = a[l];

	} else {

		build(l, mid, ls);

		build(mid + 1, r, rs);

		t[i] = t[ls] + t[rs];

	}

}

node query(int l, int r, int i, int a, int b) {

	if (a <= l && r <= b) return t[i];

	if (b <= mid) return query(l, mid, ls, a, b);

	if (mid < a) return query(mid + 1, r, rs, a, b);

	return query(l, mid, ls, a, b) + query(mid + 1, r, rs, a, b);

}

void solve() {

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

	}

	build(1, n, 1);

	cin >> q;

	for (int i = 1, l, r; i <= q; i++) {

		cin >> l >> r;

		node tmp = query(1, n, 1, l, r);

		int ans = 1e18;

		for (int i = 1; i <= tmp.tot; i++) {

			for (int j = i + 1; j <= tmp.tot; j++) {

				ans = min(ans, tmp.minn[i] | tmp.minn[j]);

			}

		}

		cout << ans << '\n';

	}

	return; 

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> T;

	while (T--) solve();

	return 0;

}
```

## Luogu P3293



多次询问求 $\max_{l\le i\le r}b \oplus(a_i+x)$ 。



对于区间异或最大值，自然而然可以想到可持久化 01 Trie 的做法，但是多了一个整体加，这个操作在可持久化 01 Trie 中实现会不太好写。



然而 01 Trie 中一个结点的本质是维护了一个连续的长为 $2$ 的整数次幂的值域，而 $+x$ 这个操作就是对值域的右移，可以通过左移查询来实现，这只需要一个可持久化值域线段树即可。



模拟 01 Trie 从高位开始贪心，目前维护的值域为 $[l-x,r-x]$ 然后贪心地根据 $b$ 的当前位和值域内是否有数来决定当前位即可。





:::info[代码]

```cpp

#include <bits/stdc++.h>

#define int long long

#define fi first

#define se second

#define mid (l + r >> 1)

using namespace std;

const int maxn = 1e5 + 5, V = (1 << 20) - 1;

int n, m, a[maxn], rt[maxn], idx;

struct node { int sum, ls, rs; } t[maxn << 5];

int build(int l, int r) {

	int u = ++idx;

	t[u].sum = 0;

	if (l == r) return u;

	t[u].ls = build(l, mid);

	t[u].rs = build(mid + 1, r);

	return u;

}

int update(int u, int l, int r, int pos) {

	int v = ++idx; t[v] = t[u], t[v].sum++;

	if (l == r) return v;

	if (pos <= mid) t[v].ls = update(t[u].ls, l, mid, pos);

	else t[v].rs = update(t[v].rs, mid + 1, r, pos);

	return v;

}

bool find(int u, int v, int l, int r, int a, int b) {

	if (t[v].sum - t[u].sum < 1) return false;

	if (a <= l && r <= b) return true;

	bool ret = false;

	if (a <= mid) ret |= find(t[u].ls, t[v].ls, l, mid, a, b);

	if (mid < b) ret |= find(t[u].rs, t[v].rs, mid + 1, r, a, b);

	return ret;

}

int solve(int b, int x, int l, int r) {

	int sum = 0;

	for (int i = 19; i >= 0; i--) {

		int L, R, op;

		if (b & (1 << i)) L = sum, R = sum + ((1 << i) - 1), op = 0;

		else L = sum + (1 << i), R = sum + (1 << (i + 1)) - 1, op = 1;

		if (!find(rt[l - 1], rt[r], 0, V, max(L - x, 0ll), min(R - x, V))) op ^= 1;

		sum |= op << i;

	} 

	return sum ^ b;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n >> m;

	rt[0] = build(0, V);

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

		rt[i] = update(rt[i - 1], 0, V, a[i]);

	}

	for (int i = 1, b, x, l, r; i <= m; i++) {

		cin >> b >> x >> l >> r;

		cout << solve(b, x, l, r) << '\n';

	}

	return 0;

}
```

# Luogu P6623



给定一棵树，其中结点 $u$ 有权值 $v_u$ ，其子树内有 $k$ 个结点 $c_k$ ，则 $val(u)=\oplus_{i=1}^{k}(v_{c_i}+dis(u,c_i))$ 。



这个不明状物是可以用 01 Trie 合并维护异或和的，只要维护每个结点，然后自下而上做合并即可。但是注意到其中有一个 $dis(u,c_i)$ ，注意到从子结点 $v$ 转移到父结点 $u$ 时所有的 $dis$ 都会 $+1$ ，问题变成怎么整体使 01 Trie 中的元素 $+1$ 。



整体 $+1$ 的操作时，可以看作当前结点 $\oplus1$ 然后判断是否进位，于是考虑从低到高位维护 01 Trie ，然后 $+1$ 操作可以看作先将某位数字交换左右子树，若进位，则递归求解。



:::info[代码]

```cpp

#include <bits/stdc++.h>

#define int long long

#define fi first

#define se second

using namespace std;

const int maxn = 5.3e5 + 5;

int n, v[maxn], p, ans, idx, rt[maxn];

vector <int> G[maxn];

struct node {

	int dep, sum, val, ch[2];

} t[maxn << 6];

void pushup(int u) {

	t[u].sum = t[t[u].ch[0]].sum + t[t[u].ch[1]].sum;

	t[u].val = t[t[u].ch[0]].val ^ t[t[u].ch[1]].val;

	if (t[t[u].ch[1]].sum & 1) t[u].val ^= 1 << t[u].dep;

}

void insert(int &u, int x, int d) {

	if (!u) u = ++idx, t[u].dep = d;

	if (d == 26) return t[u].sum++, void();

	insert(t[u].ch[(x >> d) & 1], x, d + 1);

	return pushup(u), void();

}

void change(int u) {

	if (!u) return;

	swap(t[u].ch[0], t[u].ch[1]);

	change(t[u].ch[0]);

	return pushup(u), void();

}

void merge(int &u, int &v) {

	if (!u) return u = v, v = 0, void();

	if (!v) return;

	t[u].sum += t[v].sum, t[u].val ^= t[v].val;

	merge(t[u].ch[0], t[v].ch[0]);

	merge(t[u].ch[1], t[v].ch[1]);

	return;

}

void dfs(int u) {

	rt[u] = ++idx;

	for (int v : G[u]) {

		dfs(v);

		merge(rt[u], rt[v]);

	}

	change(rt[u]);

	insert(rt[u], v[u], 0);

	ans += t[rt[u]].val;

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> v[i];

	}

	for (int i = 2; i <= n; i++) {

		cin >> p;

		G[p].push_back(i);

	}

	dfs(1);

	cout << ans << '\n';

	return 0;

}
```

## Luogu P3783



一个有向图 $n$ 个点 $m$ 条边，边 $(a,b,c,d)$ 表示从 $a$ 到 $b$ 边权为 $c$ 且对应字典树上点 $d$ 代表的字符串，从 $d_1$ 的边到 $d_2$ 的边，还会有额外边权 $LCP(d_1,d_2)$ ，求从 $1$ 开始的单源最短路。



其实 LCP 就是在 Trie 树上做 LCA 用 ST 表可以 $O(1)$ 求。



先把任意两条边边变成 ```in1--c1->out1--LCP(d1,d2)--in2--c2->out2``` 的形式，这样可以跑标准的 dijkstra ，然后对于每个原图上的点，入边与出边组合得到 $O(m^2)$ 条边。



新图上的边数太大了，考虑优化建图，对于原图上的 $u$ 将所有边按 $d$ 在 Trie 上的 dfn 序先后排序，此时得到的 $d$ 满足：



$$dep_{LCA(b_i,b_j)}=\max_{k=i}^{k<j}dep_{LCA(b_k,b_{k+1})}$$



此时将 $[1,i]$ 中所有入边向 $[i+1,n]$ 中所有出边连边，边权为 $dep_{LCA(b_i,b_{i+1})}$ ，按照上面的结论，最短的那条边是我们要的，多出来的边不影响最短路，用了前后缀优化建图，边数降至 $O(m)$ 。



:::info[代码]

```cpp

#include <bits/stdc++.h>

#define fi first

#define se second

using namespace std;

const int maxn = 5e4 + 5, maxk = 2e4 + 5, maxv = 5e5 + 5, maxe = 1e6 + 5, inf = 1e18;

int T, n, m, k, tot, ans[maxn], head[maxv], dis[maxv], vis[maxv];

struct node { int a, b, c, d; } e[maxn];

struct edge { int nxt, to, val; } E[maxe];

vector <int> in[maxn], out[maxn];

inline void add_edge(int u, int v, int w) {

	E[++tot] = {head[u], v, w};

	head[u] = tot;

}

namespace Trie {

	int tot, dfn[maxk << 1], ide[maxk], dep[maxk];

	int lg[maxk << 1], st[maxk << 1][20];

	vector <int> G[maxn];

	void build() {

		for (int i = 1; i <= k; i++) G[i].clear();

		tot = 0, dep[0] = -1;

		for (int i = 1, u, v, w; i < k; i++) {

			cin >> u >> v >> w;

			G[u].push_back(v);

			G[v].push_back(u);

		}

		auto dfs = [&](auto &&self, int u, int fa)->void{

			dep[u] = dep[fa] + 1, dfn[++tot] = u, ide[u] = tot;

			for (int v : G[u]) if (v != fa) {

				self(self, v, u);

				dfn[++tot] = u;

			}

			return;

		};

		dfs(dfs, 1, 0), lg[0] = -1;

		for (int i = 1; i < (k << 1); i++) {

			st[i][0] = dep[dfn[i]];

			lg[i] = lg[i >> 1] + 1;

		}

		for (int j = 1; j < 20; j++) {

			for (int i = 1; i + (1 << j) - 1 < (k << 1); i++) {

				st[i][j] = min(st[i][j - 1], st[i + (1 << (j - 1))][j - 1]);

			}

		}

		return;

	}

	inline int deplca(int u, int v) {

		int l = min(ide[u], ide[v]), r = max(ide[u], ide[v]);

		int k = lg[r - l + 1];

		return min(st[l][k], st[r - (1 << k) + 1][k]);

	}

	inline bool cmp(pair <int, int> a, pair <int, int> b) {

		return ide[a.fi] < ide[b.fi];

	}

};

struct temp {

	int to, val;

	friend bool operator < (temp a, temp b) {

		return a.val > b.val;

	}

};

priority_queue <temp> que;

void dijkstra(int st) {

	dis[st] = 0, que.push({st, 0});

	while (que.size()) {

		int u = que.top().to; que.pop();

		if (vis[u]) continue; vis[u] = true;

		for (int i = head[u]; i; i = E[i].nxt) {

			int v = E[i].to, w = E[i].val;

			if (dis[v] > dis[u] + w) {

				dis[v] = dis[u] + w;

				que.push({v, dis[v]});

			}

		}

	}

	return;

}

int prein[maxn << 1], preou[maxn << 1], sufin[maxn << 1], sufou[maxn << 1];

void solve() {

	cin >> n >> m >> k;

	tot = 0;

	int S = 500005, idx = m << 1;

	for (int i = 1; i <= S; i++) {

		head[i] = vis[i] = 0;

		dis[i] = inf;

	}

	for (int i = 1; i <= n; i++) {

		in[i].clear();

		out[i].clear();

	}

	for (int i = 1; i <= m; i++) {

		cin >> e[i].a >> e[i].b >> e[i].c >> e[i].d;

		out[e[i].a].push_back(i);

		in[e[i].b].push_back(i);

	}

	Trie::build();

	for (int i = 1; i <= m; i++) {

		add_edge(i, i + m, e[i].c);

		if (e[i].a == 1) add_edge(S, i, 0);

	}

	for (int i = 1; i <= n; i++) {

		vector <pair <int, int>> vec;

		for (int p : in[i]) vec.push_back({e[p].d, p + m});

		for (int p : out[i]) vec.push_back({e[p].d, p});

		sort(vec.begin(), vec.end(), Trie::cmp);

		for (int j = 0; j < vec.size(); j++) {

			prein[j] = ++idx;

			preou[j] = ++idx;

			sufin[j] = ++idx;

			sufou[j] = ++idx; 

		}

		for (int j = 0; j + 1 < vec.size(); j++) {

			add_edge(prein[j + 1], prein[j], 0);

			add_edge(preou[j], preou[j + 1], 0);

			add_edge(sufin[j], sufin[j + 1], 0);

			add_edge(sufou[j + 1], sufou[j], 0);

		}

		for (int j = 0; j < vec.size(); j++) {

			if (vec[j].se <= m) {

				add_edge(sufin[j], vec[j].se, 0);

				add_edge(prein[j], vec[j].se, 0);

			} else {

				add_edge(vec[j].se, sufou[j], 0);

				add_edge(vec[j].se, preou[j], 0);

			}

		}

		for (int j = 0; j + 1 < vec.size(); j++) {

			add_edge(sufou[j + 1], prein[j], Trie::deplca(vec[j].fi, vec[j + 1].fi));

			add_edge(preou[j], sufin[j + 1], Trie::deplca(vec[j].fi, vec[j + 1].fi));

		}

	}

	dijkstra(S);

	for (int i = 1; i <= n; i++) ans[i] = inf;

	for (int i = 1; i <= m; i++) {

		ans[e[i].b] = min(ans[e[i].b], dis[i + m]);

	}

	for (int i = 2; i <= n; i++) cout << ans[i] << '\n';

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> T; while (T--) solve();

	return 0;

}
```
