## A - Recovering BST



给定一个长为 $n$ 有序数列 $a$ 作为一个 BST 的中序遍历的节点点权，是否存在这样的 BST 使得两两相连的点权不互素。



$n\le 700,a_i\le10^9$



一个 BST 的中序遍历中，一棵子树一定是一段区间，很容易想到区间 DP 。一开始设计状态为 $f[l][r][u]$ 是区间 $[l,r]$ 以 $u$ 为根的子树是否存在，此时枚举区间和根复杂度达到 $O(n^3)$ ，此时还要判断根的左右子树是否存在根与 $u$ 相连，使用 bitset 优化可以达到 $O(\frac{n}{\omega})$ 的复杂度，但总复杂度不够优秀。



但是一个区间 $[l,r]$ 为子树，它的根的父亲一定是 $l-1$ 或 $r+1$ ，优化状态为 $f[l][r][0/1]$ 表示以 $[l,r]$ 为子树的区间的根的父亲是否可以为左/右一个节点，只需枚举区间和根，此时转移时间复杂度为 $O(n^3)$ 。



两两节点间互素关系先预处理出来，否则总时间复杂度还要加一个 log 。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 705;

int n, a[maxn];

bool f[maxn][maxn][2], g[maxn][maxn];

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n;

	for (int i = 1; i <= n; i++) cin >> a[i];

	for (int i = 1; i <= n; i++) {

		for (int j = 1; j <= n; j++) {

			g[i][j] = (__gcd(a[i], a[j]) != 1);

		}

	}

	for (int i = 1; i <= n; i++) {

		if (i != 1 && g[i][i - 1]) f[i][i][0] = true;

		if (i != n && g[i][i + 1]) f[i][i][1] = true;

	}

	for (int len = 2; len <= n; len++) {

		for (int l = 1, r = len; r <= n; l++, r++) {

			for (int k = l; k <= r; k++) {

				if ((k == l && f[l + 1][r][0]) || (k == r && f[l][r - 1][1]) || (k != l && k != r && f[l][k - 1][1] && f[k + 1][r][0])) {

					if (l == 1 && r == n) {

						cout << "Yes\n";

						return 0;

					}

					if (l != 1 && g[k][l - 1]) f[l][r][0] = true;

					if (r != n && g[k][r + 1]) f[l][r][1] = true;

				}

			}

		}

	}

	cout << "No\n";

	return 0;

}

```



## B - Yet Another Array Counting Problem



多组询问，给定一个长为 $n$ 的序列 $a$ ，求一个长为 $n$ 值域为 $[1,m]$ 的序列 $b$ 满足其每一个区间最左边最大值的位置与对应 $a$ 的区间的最左最大值位置一致，求序列 $b$ 的数量对 $998244353$ 取模。



$\sum nm\le 10^6$



先考虑怎样的序列满足每个区间的最左最大值位置与 $a$ 一致。在 $a$ 的区间 $[l,r]$ 中找到最左最大值位置 $p$ 并以其为节点，处理 $[l,p-1]$ 和 $[p+1,r]$ 为左右儿子，形成一颗 BST 满足左子树一定比该节点小，右子树小于等于该节点，若像这样处理 $b$ 的 BST 与 $a$ 的 BST 一致，则 $b$ 的所有最左最大值位置与 $a$ 一致。



设状态 $f[u][k]$ 为节点 $u$ 小于等于 $k$ 的方案总数，对于空节点的值应该为 $1$ ，叶子节点的值为 $k$ ，容易得到转移方程：



$$f[u][k] = f[u][k-1]+f[L][k-1]\times f[R][k]$$



其中 $L$ 和 $R$ 表示左右节点，时间复杂度为 $O(nm)$ 。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxdp = 1e6 + 5, maxn = 2e5 + 5, p = 1e9 + 7;

int T, n, m, root, a[maxn], lg[maxn], st[maxn][20], po[maxn][20], f[maxdp], tr[maxn][2];

inline int id(int i, int j) {

	if (!i) return 0;

	return (i - 1) * m + j;

}

int build(int l, int r) {

	if (l > r) return 0;

	int k = lg[r - l + 1], pos;

	if (st[l][k] >= st[r - (1 << k) + 1][k]) pos = po[l][k];

	else pos = po[r - (1 << k) + 1][k];

	tr[pos][0] = build(l, pos - 1);

	tr[pos][1] = build(pos + 1, r);

	return pos;

}

void dp(int u) {

	if (!u) return;

	if (tr[u][0]) dp(tr[u][0]);

	if (tr[u][1]) dp(tr[u][1]);

	if (!tr[u][0] && !tr[u][1]) {

		for (int j = 1; j <= m; j++) {

			f[id(u, j)] = j;

		}

		return;

	}

	f[id(u, 1)] = (tr[u][0] == 0) * f[id(tr[u][1], 1)];

	for (int i = 2; i <= m; i++) {

		f[id(u, i)] = (f[id(u, i - 1)] + f[id(tr[u][0], i - 1)] * f[id(tr[u][1], i)] % p) % p;

	}

	return;

}

void solve() {

	cin >> n >> m; lg[0] = -1;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

		st[i][0] = a[i];

		po[i][0] = i;

		lg[i] = lg[i >> 1] + 1;

	}

	for (int j = 1; j < 20; j++) {

		for (int i = 1; i + (1 << j) - 1 <= n; i++) {

			if (st[i][j - 1] >= st[i + (1 << (j - 1))][j - 1]) {

				st[i][j] = st[i][j - 1];

				po[i][j] = po[i][j - 1];

			} else {

				st[i][j] = st[i + (1 << (j - 1))][j - 1];

				po[i][j] = po[i + (1 << (j - 1))][j - 1];

			}

		}

	}

	root = build(1, n);

	f[0] = 1;

	dp(root);

	cout << f[id(root, m)] << '\n';

	for (int i = 1; i <= n; i++) {

		for (int j = 1; j <= m; j++) {

			f[id(i, j)] = 0;

		}

	}

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> T;

	while (T--) solve();

	return 0;

}

```



## C - Count Binary Strings



求二进制串 $s$ 满足一些条件，当 $a_{i,j}=1$ 时 $s[i...j]$ 数字相同；当 $a_{i,j}=2$ 时 $s[i...j]$ 有不同数字。求满足的 $s$ 的数量模 $998244353$ 。



$|s|\le1000$



将限制条件简化，我们设转移状态为 $f[i][j]$ 表示前 $i$ 个数中的最长相同后缀长度为 $j$ ，当 $a_{l,r}=1$ 时，说明 $r$ 对应的最长相同后缀长度至少为 $r-l+1$ ；当 $a_{l,r}=2$ 时，说明 $r$ 对应的最长相同后缀长度至多为 $r-l$ ，综上，我们可以确定长度的上下界。



那么对于 $j=1$ 时，有 $f[i][j] = \sum_{x\in[1,i)} f[i-1][x]$ ，否则有 $f[i][j] = f[i - 1][j-1]$ ，随后将长度上下界以外的状态设为 $0$ 即可，因为第一个数可以是 $0$ 或 $1$ ，则 $f[1][1]=2$ ，最后答案为 $\sum f[n][j]$ 。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 105, p = 998244353;

int n, a[maxn][maxn], f[maxn][maxn], lim[maxn][2];

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n;

	for (int i = 1; i <= n; i++) {

		lim[i][0] = 1;

		lim[i][1] = i;

	}

	for (int i = 1; i <= n; i++) {

		for (int j = i; j <= n; j++) {

			cin >> a[i][j];

			if (a[i][j] == 1) lim[j][0] = max(lim[j][0], j - i + 1);

			if (a[i][j] == 2) lim[j][1] = min(lim[j][1], j - i);

			if (i == j && a[i][j] == 2) {

				cout << "0\n";

				return 0;

			}

		}

	}

	f[1][1] = 2;

	for (int i = 2; i <= n; i++) {

		for (int j = 1; j < i; j++) {

			(f[i][1] += f[i - 1][j]) %= p;

		}

		for (int j = 2; j <= i; j++) {

			f[i][j] = f[i - 1][j - 1];

		}

		for (int j = 1; j <= i; j++) {

			if (lim[i][0] <= j && j <= lim[i][1]) continue;

			f[i][j] = 0;

		}

	}

	int ans = 0;

	for (int i = 1; i <= n; i++) {

		(ans += f[n][i]) %= p;

	}

	cout << ans << '\n';

	return 0;

}

```



## D - Binary String Wowee



给定二进制串 $s$ ，一次操作是将任意一个 $s_i=0$ 的 $[1,i]$ 反转，将进行 $k$ 次操作，问不同的操作方式有多少。



从后往前考虑没有后效性，当在 $i$ 进行操作时，若 $s_i=0$ 则在 $(i,n]$ 上有偶数次操作，否则有奇数次操作。设 $f[i][j]$ 为在 $[i,n]$ 上做了 $j$ 次操作的方案数。设在 $i$ 做了 $x$ 次操作，那么保持在 $(i,n]$ 上的 $j$ 次操作相对位置一致，在偶/奇数位置插入 $x$ 个操作，细说：



若 $s_i=0$ 则 $f[i][j]=\sum f[i+1][j-x]\times C_{\lceil j\rceil}^{x}$



若 $s_i=1$ 则 $f[i][j]=\sum f[i+1][j-x]\times C_{\lfloor j\rfloor}^{x}$



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 1005, p = 998244353;

int T, n, k, f[maxn][maxn], C[maxn][maxn];

char s[maxn];

int ksm(int a, int b) {

	int res = 1;

	while (b) {

		if (b & 1) res = res * a % p;

		a = a * a % p, b >>= 1;

	}

	return res;

}

void solve() {

	cin >> n >> k;

	for (int i = 1; i <= n; i++) cin >> s[i];

	for (int i = 0; i <= n + 1; i++) for (int j = 0; j <= k; j++) f[i][j] = 0;

	f[n + 1][0] = 1;

	for (int i = n + 1; i > 1; i--) {

		for (int j = 0; j <= k; j++) {

			for (int x = 0; j + x <= k; x++) {

				if (s[i - 1] == '1') (f[i - 1][j + x] += f[i][j] * C[(j + x) >> 1][x] % p) %= p;

				else (f[i - 1][j + x] += f[i][j] * C[(j + x + 1) >> 1][x] % p) %= p;

			}

		}

	}

	cout << f[1][k] << '\n';

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> T;

	C[0][0] = 1;

	for (int i = 1; i <= 1000; i++) {

		for (int j = 0; j <= i; j++) {

			if (i == j || j == 0) C[i][j] = 1;

			else C[i][j] = (C[i - 1][j] + C[i - 1][j - 1]) % p;

		}

	}

	while (T--) solve();

	return 0;

}

```



## E - Jongmah



给你颜色在 $[1,m]$ 内的牌共 $n$ 个，可以取出 $(i-1,i,i+1)$ 和 $(i,i,i)$ 问最多可以去多少个。一个 $(i-1,i,i+1)$ 最多出现两次，否则会变为 $(i,i,i)$ 处理，设 $f[i][a][b]$ 为前 $i$ 个有 $a$ 个 $(i-1,i,i+1)$ 和 $b$ 个 $(i,i+1,i+2)$ 。



$$f[i][b][c] = \max(f[i-1][a][b]+c+\lfloor\frac{cnt_i-a-b-c}{3}\rfloor)$$



答案显然为 $f[m][0][0]$ 。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 1e6 + 5;

int n, m, ans, a[maxn], cnt[maxn], f[maxn][3][3];

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n >> m;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

		cnt[a[i]]++;

	}

	for (int i = 0; i <= m; i++) for (int a = 0; a < 3; a++) for (int b = 0; b < 3; b++) for (int c = 0; c < 3; c++) {

		if (cnt[i + 1] >= a + b + c) f[i + 1][b][c] = max(f[i + 1][b][c], f[i][a][b] + c + (cnt[i + 1] - a - b - c) / 3);

	}

	cout << f[m][0][0] << '\n';

	return 0;

}

```



## F - Almost Triple Deletions



给定长为 $n$ 的数组 $a$ ，一次操作可以选相邻不同数删除，最后最多可以留下多少相同的数。



这样的区间可以被完全删除：长度为偶数且众数数量不超过区间长度的一半，证明略。我们预处理每一个区间是否可以被完全删除，然后定义 $f[i]$ 是前 $i$ 个数中可以留下多少个 $a_i$ ，通过前面与 $a_i$ 值相等的数进行转移即可，加一写特判，如：第一个这样的数要满足前面的东西都可以被删掉，最后的数要满足后面的数都可以被删掉。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 5005;

int T, n, a[maxn], f[maxn], cnt[maxn]; 

bool ok[maxn][maxn], L[maxn], R[maxn]; 

void solve() {

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

	}

	memset(ok, 0, sizeof ok);

	memset(f, 0, sizeof f);

	memset(L, 0, sizeof L);

	memset(R, 0, sizeof R);

	for (int l = 1; l <= n; l++) {

		int mx = 0;

		memset(cnt, 0, sizeof cnt);

		for (int r = l; r <= n; r++) {

			mx = max(mx, ++cnt[a[r]]);

			if ((r - l + 1) & 1) ok[l][r] = false;

			else ok[l][r] = ((mx << 1) <= (r - l + 1)); 

		}

	}

	for (int i = 1; i <= n; i++) {

		if (i == 1 || (i != 1 && ok[1][i - 1])) f[i] = 1;

		for (int j = 1; j < i; j++) {

			if (a[i] != a[j]) continue;

			if (!f[j]) continue;

			if (j + 1 == i || ok[j + 1][i - 1]) f[i] = max(f[i], f[j] + 1); 

		}

	}

	int ans = 0;

	for (int i = 1; i <= n; i++) {

		if (i == n || (i != n && ok[i + 1][n])) ans = max(ans, f[i]);

	}

	cout << ans << '\n';

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> T;

	while (T--) solve();

	return 0;

}

```



## G - Fuel Round Trip



数轴上从 $a_1$ 出发到 $a_n$ 再回到 $a_1$ ，经过一个单位长度耗一升油，有 $n$ 个加油站为 $a$ 数组，第 $i$ 个加油站可花费 $p_i$ 加 $f_i$ 升油，往返不得在同一个加油站加油，油箱上限为 $h$ 升，问最小花费为多少。



因为有后效性不好做，所以设计状态时要把往返状态都加进去，这里令 $F[i][a][b]$ 为从 $1$ 到 $i$ 有 $a$ 升油，回来有 $b$ 升，根据题意转移即可，注意边界条件。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 305;

int n, h, F[maxn][maxn][maxn], a[maxn], p[maxn], f[maxn];

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n >> h;

	for (int i = 1; i <= n; i++) cin >> a[i];

	for (int i = 1; i < n; i++) cin >> p[i] >> f[i];

	memset(F, 0x3f, sizeof F);

	F[0][h][0] = 0;

	for (int i = 1; i <= n; i++) {

		int dis = a[i] - a[i - 1];

		for (int j = 0; j <= h; j++) for (int k = 0; k <= h; k++) {

			int J = j - dis, K = k + dis;

			if (J < 0 || K > h) continue;

			F[i][J][K] = min(F[i - 1][j][k], F[i][J][K]);

			F[i][min(h, J + f[i])][K] = min(F[i][min(h, J + f[i])][K], F[i - 1][j][k] + p[i]);

			F[i][J][max(0ll, K - f[i])] = min(F[i][J][max(0ll, K - f[i])], F[i - 1][j][k] + p[i]);

		}

	}

	int ans = 0x3f3f3f3f3f3f3f3f;

	for (int i = 0; i <= h; i++) for (int j = 0; j <= i; j++) ans = min(ans, F[n][i][j]);

	if (ans == 0x3f3f3f3f3f3f3f3f) ans = -1;

	cout << ans << '\n';

	return 0;

}

```



## H - Kaavi and Magic Spell



给定字符串 $S$ 和 $T$ 分别长为 $n$ 和 $m$ ，有一个空串 $A$ ，一次操作可以删除 $S$ 的第一个字符并添加至 $A$ 的首部或尾部，共 $n$ 次操作，使得 $T$ 是 $A$ 的前缀的方案数是多少。



设 $f[l][r]$ 为 $S$ 的前 $r-l+1$ 个字符组成 $T(l...r)$ 的方案数，边界条件是当 $T_i=S_1$ 时 $f[i][i]=2$ ，转移很显然，当 $S_{len}=T_l$ 时 $f[l][r]$ 由 $f[l+1][r]$ 得，当 $S_{len}=T_r$ 时 $f[l][r]$ 由 $f[l][r-1]$ 得。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 3005, p = 998244353;

int n, m, f[maxn][maxn];

char s[maxn], t[maxn];

signed main() {

//	ios::sync_with_stdio(0);

//	cin.tie(0); cout.tie(0);

//**CF不给cin>>(s+1) 

	scanf("%s%s", s + 1, t + 1);

	n = strlen(s + 1), m = strlen(t + 1);

	for (int i = 1; i <= n; i++) f[i][i] = (i > m || s[1] == t[i]) << 1;

	for (int len = 2; len <= n; len++) {

		for (int l = 1, r = len; r <= n; l++, r++) {

			if (l > m || s[len] == t[l]) (f[l][r] += f[l + 1][r]) %= p;

			if (r > m || s[len] == t[r]) (f[l][r] += f[l][r - 1]) %= p;

		}

	}

	int ans = 0;

	for (int i = m; i <= n; i++) (ans += f[1][i]) %= p;

	cout << ans << '\n';

	return 0;

}

```

## A 书架平衡



有 $n$ 个数对 $u_i$ 和 $d_i$ 以及一个整数 $k$ ，花费一个金币可以让任意一个数减一，问最少花费多少可以使：



1. $u_i+d_i=h$ 其中 $h$ 是定值

2. $|u_i-u_{i-1}|\le k$ 即相邻的 $u$ 之间差值不大于 $k$



乍看没什么思路，很多东西也没有特殊限制，不妨找一下，发现第二个条件有点难调，但是由于只能减数，于是可以确定每对数减多少，发现 $h$ 的合法性是单调的，于是我们使用二分答案得到最大的合法 $h$ ，答案即为 $\sum(u_i+d_i)-nh$ 。



如何检查是否可以使得差值不大于 $k$ 呢，因为 $u_i+d_i$ 整体要减为 $h$ ，因此可以推导出 $u_i$ 减后的上下界。设当前的上下界为 $[l,r]$ 随后将其扩展至 $[l-k,r+k]$ 与下一个 $u_i$ 的上下界去并，重复操作，若最后得到的上下界是空集则检查不通过。



综上检查 $O(n)$ 在二分的基础上时间复杂度为 $O(n\log n)$ 而 $n=2\times 10^5$ 可以通过。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 2e5 + 5;

int n, x, minn = 0x3f3f3f3f3f3f3f3f, sum, u[maxn], d[maxn];

int f[maxn][2], mx[maxn], mn[maxn];

bool check(int h) {

	for (int i = 1; i <= n; i++) {

		int tmp = u[i] + d[i] - h;

		mx[i] = u[i] - max(tmp - d[i], 0ll);

		mn[i] = max(u[i] - tmp, 0ll);

	}

	int l = mn[1], r = mx[1];

	if (l > r) return false;

	for (int i = 2; i <= n; i++) {

		l -= x, r += x;

		l = max(l, mn[i]);
```

## B 宴会后厨



有 $n$ 道菜，每道菜所需时间为 $a_i$ 小时，一道菜至少要 $k$ 个厨师参与。有 $m$ 个厨师，每个人可以花费 $b_i$ 小时，同时无论如何只要厨师工作就一定会支付 $b_i$ 的工资。求最少浪费的工资，即要使厨师不工作但得到工资的时间最少，并输出这个最小值，详细来讲，设选的厨师的集合为 $S$ 则答案为 $\sum_{i\in S} b_i-\sum a_i$ ，不可能则输出 ```Impossible``` 。



先判断无解的情况，显然 $a_i<k$ 的情况是无解的。一道菜可以拆分为 $k$ 个不同人贡献的基础小时和 $a_i-k$ 人贡献的额外小时，显然一个厨师的基础小时最多为 $\min(n,b_j)$ 且厨师贡献的基础小时之和一定不小于每道菜的厨师种类之和即 $nk\le\sum\min(n,b_j)$ 。啪一下变成了 01 背包问题了，设 $f_{i,j}$ 为前 $i$ 个厨师总雇佣时间为 $j$ 的最大基础小时之和，做 01 背包加滚动数组即可，因为 $a_i$ 和 $b_i$ 和 $n$ 同阶，时间复杂度为 $O(n^3)$ 而 $n\le300$ 可以通过。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 305;

int n, m, k, sumA, sumB, a[maxn], b[maxn], f[maxn * maxn];

signed main() {

		if (a[i] < k) {

			cout << "Impossible\n";

			return 0;

		}

		sumA += a[i];

	}

	for (int i = 1; i <= m; i++) {

		cin >> b[i];

		sumB += b[i];

	}

	memset(f, -0x3f, sizeof f);

	f[0] = 0;

	for (int i = 1; i <= m; i++) {

		int val = min(n, b[i]);

		for (int j = sumB - b[i]; j >= 0; j--) {

			if (f[j] < 0) continue;

			f[j + b[i]] = max(f[j + b[i]], f[j] + val);

		}

	}

	for (int j = sumA; j <= sumB; j++) {

		if (f[j] >= n * k) {

			cout << j - sumA << '\n';

		}

	}
```

## C 边防军阵



给定一个 $n$ 个节点的树和 $m$ 个点对 $(u_i,v_i)$ 。对于每个点对可以在这个点对之间的简单路径上的边打标记，或在除了点对之间的简单路径上的边打标记，求最后有标记的边数的最小值。



正难则反，我们求没有标记的边数的最大值 $M$ 则答案为 $n-1-M$ 。任意两条边只有被标记的的状态相同才能一起不被覆盖，状态相同的边集的大小的最大值即为 $M$ 。现在考虑如何标记/链修改状态，使用 XorHash 可以做到，只要做类似树上差分的东西就可以，将每一个边的哈希值向上合并，最后用一个 umap 统计即可。注意：出题人很闲会卡 mt19937_64 的哈希。



```cpp

#include <bits/stdc++.h>

#define int long long

#define ull unsigned long long

using namespace std;

const int maxn = 1e6 + 5;

ull seed = time(0);

ull splitmix64(ull x) {

	x += 0x9e3779b97f4a7c15ULL;

	x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;

	x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;

	return x ^ (x >> 31);

}

ull get_hash(ull id) {

	return splitmix64(seed + id * 0x9e3779b97f4a7c15ULL);

}

ull val[maxn], hsh[maxn];

unordered_map <ull, int> mp;

void dfs(int u, int fa) {

	for (auto g : G[u]) {

		if (g == fa) continue;

		dfs(g, u);

		val[u] ^= val[g];

	}

	return;

}
```

## A - 「KWOI R2」染色



将长度为 $n$ 的序列 $a$ 拆分为两个子序列使得这两个子序列的相邻两个数的差的绝对值的和最大。



显然可以列出 DP 状态 $f[i]$ 表示前 $i$ 个中的最大值，设 $s[i]=\sum_{x=2}^{x\le i}|a[i]-a[i-1]|$ 则有 $f[i] = \max(f[j] + |a[i] - a[j]| + s[i - 1] - s[j])$ 。



有绝对值不好做，但是有一个相关 trick ： $|a-b|=\max(a-b,b-a)$ 可以将式子转化为 $\max$ 套 $\max$ 化简变成 $f[i]= \max(\max(f[j] - a[j] - s[j]) + (s[i - 1] + a[i]),\ max(f[j] + a[j] - s[j]) + (s[i - 1] - a[i]))$



其中的 $\max(f[j]-a[j]-s[j])$ 和 $\max(f[j]+a[j]-s[j])$ 使用前缀最大值维护。

## B - Discharging



将一个长为 $n$ 的序列分组，第 $i$ 个人等待的时间是当前组和前面的所有组中的最大值之和，求所有人等待时间最小值。



$f[i]$ 为前 $i$ 个中分完组的时间最大值

在 $[l,r]$ 中最大值为 $k$ 则 $a[r + 1] > k$ 否则 $[l,r+1]$ 更优。



使 $b[i] = \max(a[1..i])$ , 则列出 $f[i] = max(f[j] + (n-j) \times b[i])$ 。



设决策点为 $j$ 得：

```

f[i] - n * b[i] = f[j] - j * b[i] 

f[j] = b[i] * j + (f[i] - n * b[i])

   y =  k   * x +   b

```

在上凸包中试斜率为 $k$ 的直线，截距最大的即为所求 

用单调栈维护上凸包，用一个指针在栈中寻找斜率最正确的 $j$ 即可。



```cpp

#include <bits/stdc++.h>

#define int long long

#define fi first

#define se second

using namespace std;

const int maxn = 1e6 + 5;

int n, a[maxn], f[maxn], top;

pair <int, int> st[maxn];

double slope(pair <int, int> a, pair <int, int> b) {

	return double(a.se - b.se) / double(a.fi - b.fi);

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

		a[i] = max(a[i], a[i - 1]);

	}

	st[++top] = {0, 0};

	int j = 1;

	for (int i = 1; i <= n; i++) {

		while (j < top && a[i] > slope(st[j + 1], st[j])) j++;

		while (j > 1 && a[i] < slope(st[j - 1], st[j])) j--;

		int j0 = st[j].fi;

		f[i] = f[j0] + (n - j0) * a[i];

		while (top > 1 && slope({i, f[i]}, st[top]) < slope(st[top], st[top - 1])) top--;

		st[++top] = {i, f[i]}, j = min(j, top);

	}

	cout << f[n] << '\n';

    return 0;

}
```

## C - Vasya and Array



一个长度为 $n$ 的数列，每个数为 $[1,k]$ 中的整数或 $-1$ 且 $-1$ 可以被更改为任意 $[1,k]$ 中的整数，求最后不存在连续 $len$ 个相同数字的操作方案数。



$n\le10^5,k\le100$



首先列出状态 $f[i][j]$ 表示第 $i$ 个位置放 $j$ 的合法方案数，设 $s[i] = \sum f[i][k]$ 。



先不考虑 $len$ 的限制：

1. $a[i]=j$ 则可以接在所有 $i-1$ 后面

2. $a[i]=-1$ 可以填 $[1,k]$ 任意数字

3. 其他情况方案数为 $0$ 。



则 ```f[i][j] = (a[i] == -1 || a[i] == j) ? s[ - 1] : 0``` 。



然后减去填上 $j$ 后第一次形成连续 $len$ 个 $j$ 的非法方案数，在每一个位置都排除首次非法方案可以避免重复统计。前 $i-1$ 个都是合法的，只有可能是第 $i$ 个是第 $len$ 个 $j$ 。



此时非法方案为 $\sum_{c\neq j}f[i-len][c] = s[i-len] - f[i-len][j]$ 。同时要用前缀和判断 $[]i-len+1,i$ 中是否只有 $j$ 和 $-1$ 。



时间复杂度 $O(nk)$ 。



```cpp

#include <bits/stdc++.h>

#define int long long 

using namespace std;

const int maxn = 1e5 + 5, maxk = 105, p = 998244353;

struct mint {

    int val;

    static int norm(const int& x) { return x < 0 ? x + p : x; }

    mint() : val(0) {}

    mint(const int& m) : val(norm(m % p)) {}

    mint operator - () const { return mint(norm(-val)); }

    bool operator == (const mint& o) { return val == o.val; }

    mint& operator += (const mint& o) { return val = (val + o.val) % p, *this; }

    mint& operator -= (const mint& o) { return val = norm(val - o.val), *this; }

    mint& operator *= (const mint& o) { return val = static_cast <int>(val * o.val % p), *this; }

    mint operator - (const mint& o) const { return mint(*this) -= o; }

    mint operator + (const mint& o) const { return mint(*this) += o; }

    mint operator * (const mint& o) const { return mint(*this) *= o; }

    friend std::istream& operator >> (std::istream& is, mint& a) {}

    friend std::ostream& operator << (std::ostream& os, const mint& a) { return os << a.val; }

};

int n, k, len, a[maxn], sum[maxn][maxk];

mint f[maxn][maxk], s[maxn];

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n >> k >> len;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

	}

	for (int i = 1; i <= n; i++) {

		for (int j = 1; j <= k; j++) {

			if (a[i] == j || a[i] == -1) {

				sum[i][j] = sum[i - 1][j] + 1;

			}

		}

	}

	s[0] = 1;

	for (int i = 1; i <= n; i++) {

		for (int j = 1; j <= k; j++) {

			if (a[i] == j || a[i] == -1) {

				f[i][j] = s[i - 1];

				if (sum[i][j] >= len) {

					f[i][j] += f[i - len][j] - s[i - len];

				}

				s[i] += f[i][j];

			}

		}

	}

	cout << s[n] << '\n';

	return 0;

}
```

## E - Patisserie ABC 3



有 $n$ 个蛋糕，第 $i$ 个蛋糕有属性 $(x_i,y_i,z_i)$ ，选取 $k$ 对蛋糕，一对蛋糕的美味值为 $\max(x_i+x_j,y_i+y_j,z_i+z_j)$ 求总美味值最大为多少。



首先发现一坨 $\max(x_i+x_j,y_i+y_j,z_i+z_j)$ 很难处理，于是想到让每一个蛋糕自主选择哪个值参与配对，那么选了哪些蛋糕就不重要了，重要的变为目前三种值存在的奇偶性，若是奇数，则当前蛋糕可以配对，最后在动态规划过程中求得的最大值即为所求，于是自然而然地列出状态：用 $f_{i,j,mask}$ 表示前 $i$ 个蛋糕中选了 $j$ 个蛋糕，目前三种状态的奇偶性为 $mask$ ，其中 $mask$ 是一个三位二进制数。



显然可以分情况，当前不选，当前选用 $x$ 或 $y$ 或 $z$ 值配对，即：



```cpp

f[i][j][mask] = max(f[i - 1][j - 1][mask], f[i - 1][j - 1][mask ^ 1] + x[i], f[i - 1][j - 1][mask ^ 2] + y[i], f[i - 1][j - 1][mask ^ 4] + z[i])

```



答案为 $f_{n,2k,0}$



时间复杂度 $O(nk)$ 不能通过。



设选 $i$ 个蛋糕最后的最大值为 $F(i)$ 并可以发现以下特征：



1. 题目中要求在 $n$ 个蛋糕中选 $2k$ 个。

2. 如果没有 $2k$ 个的限制会很容易求，只需要把 DP 的第二维去掉即可。

3. 函数 $F(i)$ 有凸性，显然的，设 $F(x_0)$ 是最大值，则 $|i-x_0|$ 越大，答案 $F(i)$ 只会更劣。



综上，这道题完全符合使用 wqs 二分的性质！



二分斜率即可，但是斜率可能出现 $0.5$ 所以有乘除 $2$ 的处理，同时我们还要计算当前斜率下的蛋糕数，只需在成功转移时记录即可。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 1e6 + 5;

int T, n, k, val[maxn][3], f[maxn][8], cnt[maxn][8];

void DP(int h) {

	for (int i = 0; i < 8; i++) {

		f[0][i] = -0x3f3f3f3f3f3f3f3f;

		cnt[0][i] = 0;

	}

	f[0][0] = 0;

	for (int i = 1; i <= n; i++) {

					f[i][mask] = tmp1;

					cnt[i][mask] = tmp2;

				}

			}

		}

	}

	return;

}
```

## F - 收拾背包 / Pakowanie plecaka



有 $n$ 个物品重量为 $w_i$ 对于每一个 $k$ 求出背包容量最小值 $W$ 使得从 $1$ 到 $n$ 的顺序，可以装下 $k$ 个物品。



假设选了一组物品，总重量为 $S$ ，背包总重为 $W$ ，显然最优时 $W = S$ 。当检查到 $i$ 时，背包剩余容量是之后要选取物品的重量之和，要合法地跳过物品 $i$ 则物品 $i$ 后选的物品重量之和 $< w[i]$ 。



启发从后向前 DP ：令 $f[i][j]$ 是 $[i, n]$ 的物品中选了 $j$ 的最小代价，则：



```cpp

f[i][j] = f[i + 1][j - 1] + w[i] //选第 i 件

f[i][j] = f[i + 1][j] //不选第 i 件，此时 f[i + 1][j] < w[i]

```



使用滚动数组滚掉一维，在处理第 $i$ 个物品时考虑 $f[j]$ 有什么变化，假设 $f[j] < w[i]$ 的最大下标为 $k$ （ $f[j]$ 单调递增），则：



1. $j <= k$ 时，可以合法跳过 $i$ 则 $f[1...k]$ 不变。

2. $j > k$ 时，一定要选，从 $f[j - 1]$ 整体 $+w[i]$ 转移。



对它使用平衡树吧，参考题解的实现：



```cpp

#include <bits/stdc++.h>

#define int long long 

using namespace std;

const int maxn = 5e5 + 5;

int n, w[maxn], root, ls[maxn], rs[maxn], key[maxn], val[maxn], tag[maxn];

mt19937 rnd(676767);

void apply(int u, int k) {val[u] += k, tag[u] += k;}

void pushdown(int u) {apply(ls[u], tag[u]), apply(rs[u], tag[u]), tag[u] = 0;}

void newnode(int u, int k) {val[u] = k, tag[u] = 0, key[u] = rnd();}

void split(int u, int k, int &x, int &y) {

	if (!u) return x = y = 0, void(); pushdown(u);

	if (val[u] <= k) split(rs[u], k, rs[x = u], y);

	else split(ls[u], k, x, ls[y = u]);

}

int merge(int u, int v) {

	if (!u || !v) return u | v; pushdown(u), pushdown(v);

	if (key[u] > key[v]) return rs[u] = merge(rs[u], v), u;

	else return ls[v] = merge(u, ls[v]), v;

}

int query(int u) {

	if (!u) return 0;

	while (rs[u]) pushdown(u), u = rs[u];

	return val[u];

}

void solve(int u) {

	pushdown(u);

	if (ls[u]) solve(ls[u]);

	cout << val[u] << ' ';

	if (rs[u]) solve(rs[u]);

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n;

	for (int i = 1; i <= n; i++) cin >> w[i];

	newnode(root = n, w[n]);

	for (int i = n - 1; i >= 1; i--) {

		int u, v; split(root, w[i] - 1, u, v);

		newnode(i, query(u) + w[i]), apply(v, w[i]);

		root = merge(merge(u, i), v);

	}

	solve(root);

	return 0;

}
```

## A - 01 on Tree



给定 $n$ 个节点的树，每个节点权值为 $0$ 或 $1$ 求树上节点的拓扑序使得组成序列的逆序对数最小。



不考虑拓扑序的限制，则 ```00..0111..1``` 是最优的 ，于是想到，有 $v[i] = 0$ 则一旦父亲被选了则立即选它，我们可以将其与父亲节点合并。



```fa[x]  A1  A2  A3  x```



其中 $A1$ $A2$ $A3$ 应是 $x$ 的兄弟子树，每个兄弟子树应该有一个权值，套路 Exchange Argument 考虑相邻两个块 $a, b$ ，逆序对数量 $cnt_a1 * cnt_b0 < cnt_b1 * cnt_a0$ 时，显然 $a$ 在 $b$ 前面，那么每个节点的权值为 $\frac{cnt1}{cnt0}$ 。



用 set 维护权值，每次取最小的一个与父亲节点合并，用并查集维护答案。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 2e5 + 5;

int n, fa[maxn], w[maxn], p[maxn], cnt[maxn][2], vis[maxn], ans;

struct node {

	int u, cnt0, cnt1;

	node(int a = 0, int b = 0, int c = 0) {

		u = a, cnt0 = b, cnt1 = c;

	}

	friend bool operator < (node a, node b) {

		return a.cnt1 * b.cnt0 < b.cnt1 * a.cnt0;

	}

};

multiset <node> st;

int get(int x) {

	if (p[x] == x) return x;

	return p[x] = get(p[x]);

}

void merge(int u, int v) {

	u = get(u), v = get(v);

	if (u == v) return;

	ans += cnt[u][0] * cnt[v][1];

	cnt[v][0] += cnt[u][0];

	cnt[v][1] += cnt[u][1];

	p[u] = v;

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n;

	for (int i = 2; i <= n; i++) {

		cin >> fa[i];

	}

	for (int i = 1; i <= n; i++) {

		cin >> w[i];

		p[i] = i, cnt[i][w[i]]++;

		st.insert(node(i, cnt[i][0], cnt[i][1]));

	}

	while (st.size()) {

		int u = st.begin()->u;

		st.erase(st.begin());

		if (vis[u]) continue;

		vis[u] = true;

		if (!fa[u]) continue;

		int v = get(fa[u]);

		merge(u, v);

		st.insert(node(v, cnt[v][0], cnt[v][1]));

	}

	cout << ans << '\n';

	return 0;

}
```

## B - 皇后游戏



每个大臣有两个数 $a_i$ 和 $b_i$ 求一种排序顺序使得 $c_i = max(c_{i-1}, \sum_{j\le i} a_i) + b_i$ 最大，使用 Exchange Argument 来寻找贪心规则，若相邻的 $i < j$ 是优的，设 $i$ 前的 $a$ 之和为 $x$ ， $i$ 前的 $c$ 为 $y$ 则较大的 $c$ 值：



$$\max(\max(y, x + a_i) + b_i, x + a_i + a_j) + b_j$$



$$\max(y + b_i + b_j, x + a_i + b_i + b_j, x + a_i + a_j + b_j)$$

交换 $i$ $j$ 后为：



$$\max(y + b_i + b_j, x + a_j + b_i + b_j, x + a_i + a_j + b_i) $$



按照题意：



$$\max(x + a_i + b_i + b_j, x + a_i + a_j + b_j) \le \max(x + a_j + b_i + b_j, x + a_i + a_j + b_i)$$



消去 x 得： 

$$a_i + b_j + \max(b_i, a_j) \le a_j + b_i + \max(b_j, a_i)$$

$$\max(a_j, b_i) - a_j - b_i \le \max(a_i, b_j) - a_i - b_j$$



因为 $\max(a, b) - a - b = -\min(a, b)$ 则：



$$-\min(a_j, b_i) \le -\min(a_i, b_j)$$

      

$$\min(a_i, b_j) \le \min(a_j, b_i) $$



小于等于不满足传递性，于是继续优化：



$\min(a_i, b_j) \le \min(a_j, b_i)$ 有以下三种可能：

1. $a_i < b_i$ 且 $a_j < b_j$ ，此时要按 $a$ 升序排序

2. $a_i = b_i$ 且 $a_j = b_j$ ，此时无限制

3. $a_i > b_i$ 且 $a_j > b_j$ ，此时要按 $b$ 降序排序

这样就满足传递性了。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 1e5 + 5;

int T, n;

struct node {

	int a, b, d;

	void D() {

		if (a < b) d = -1;

		if (a > b) d = 1;

		if (a == b) d = 0;

	}

	friend bool operator < (node x, node y) {

		if (x.d != y.d) return x.d < y.d;

		if (x.d <= 0) return x.a < y.a;

		return x.b > y.b;

	}

} w[maxn];

void solve() {

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> w[i].a >> w[i].b;

		w[i].D();

	}

	sort(w + 1, w + n + 1);

	int sum = 0, c = 0;

	for (int i = 1; i <= n; i++) {

		sum += w[i].a;

		c = max(c, sum) + w[i].b;

	}

	cout << c << '\n';

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> T;

	while (T--) solve();

	return 0;

}
```

## C - Coins



有 $x+y+z$ 个人，每个人有 $a_i$ $b_i$ $c_i$ 个三种不同的钱，这三种分别要 $x$ $y$ $z$ 个，求最后最大总钱数



先简化问题，三种钱太麻烦了，我们将三元组 $(a_i,b_i,c_i)$ 变为二元组 $(a_i-c_i,b_i-c_i)$ ，即默认每个人都贡献第三种钱，然后选 $x$ 个人贡献 $a_i-c_i$ 个第一种和 $y$ 个人贡献 $b_i-c_i$ 个第二种。



朴素地想，按照 $a_i-b_i$ 升序排序，这样正着选 $a$ 应该更有利，倒着选 $b$ 也更有利，但是证以下发现这不符合贪心策略，所以引入反悔操作。 



变成反悔贪心了，扫一遍数组，先加入 $a_i$ 若总数大于 $x$ 则减去先前选了的最小值，这样在第 $i$ 时刻就是最大值，我们记 $l_i$ 为前 $i$ 个 $a$ 中最大的 $x$ 个，同理倒着做 $b$ 记 $r_i$ 为后 $i$ 个中 $b$ 的最大的 $y$ 个。 



最后答案就是将序列分为两段，前一段取 $x$ 个 $a$ 最大，后一段取 $y$ 个 $b$ 最大，这样贪心地取是最优的，为什么取的不会混在一起？因为每一次取得都是最大的，易证若混在一起一定能找到不混在一起的更优解。





```cpp

#include<bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 1e5 + 5; 

priority_queue<int, vector<int>, greater<int>> q;

int x, y, z, a[maxn], b[maxn], pos[maxn], c, res, l[maxn], r[maxn];

bool cmp(int i, int j) {return a[i] - b[i] > a[j] - b[j];}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> x >> y >> z;

	for (int i = 1; i <= x + y + z; i++) {

		cin >> a[i] >> b[i] >> c;

		a[i] -= c, b[i] -= c, res += c, pos[i] = i;

	}

	sort(pos + 1, pos + x + y + z + 1, cmp);

	for (int i = 1; i <= x + y + z; i++) {

		l[i] = l[i - 1] + a[pos[i]], q.push(a[pos[i]]);

		if ((int)q.size() > x) l[i] -= q.top(), q.pop();

	}

	while (q.size()) q.pop();

	for (int i = x + y + z; i >= 1; i--) {

		r[i] = r[i + 1] + b[pos[i]], q.push(b[pos[i]]);

		if ((int)q.size() > y) r[i] -= q.top(), q.pop();

	}

	int ans = -1e18;

	for (int i = x; i <= x + z; i++) {

		ans = max(ans, l[i] + r[i + 1]);

	}

	cout << ans + res << '\n'; 

	return 0;

}

```

## E - Email Scheduling Optimization



给定数组 $a$ , $b$ 。给第 $i$ 家公司写邮件要 $a_i$ 分钟，发出后 $b_i$ 分钟得到应答，一次查询会修改 $a$ 或 $b$ 中的值，要回答最早收到所有回复的时间。 



安排一个顺序，则显然答案为 $\max(\sum_{j=1}^{i}a_j+b_i)$ ，依旧 Exchange Argument 导出 $b_i$ 从大到小做最优。



按照 $b$ 从大到小存入线段树，我们想一想区间答案： 



$$

\begin{aligned}

f_{l,r} &= \max_{i=l}^{r}(\sum_{j=l}^{r}a_j+b_i)\\

&= \max(max_{i=l}^{k}(\sum_{j=l}^{r}a_j+b_i), \max_{i=k+1}^{r}(\sum_{j=l}^{r}a_j+b_i))\\

&= \max(\max_{i=l}^{k}(\sum_{j=l}^{r}a_j+b_i), \sum_{j=l}^{k}a_j+\max_{i=k+1}^{r}(\sum_{j=k + 1}^{r}a_j+b_i))\\

&= \max(f_{l,k}, f_{k+1,r}+\sum_{j=l}^{k}a_j)

\end{aligned}

$$



可以区间合并，用线段树维护即可，注意 $b$ 是降序的所以线段树的合并要反过来，应该是：



$$ans_u=\max(ans_{ls}+sum_{rs},ans_{rs})$$



```cpp

#include<bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 1e6 + 5;

int n, q, tot, lsh[maxn << 1], a[maxn], b[maxn], op[maxn], id[maxn], x[maxn], sum[maxn << 2], ans[maxn << 2], cnt[maxn << 2];

inline int find(int x) {

	return lower_bound(lsh + 1, lsh + tot + 1, x) - lsh;

} 

#define ls (i << 1)

#define rs (ls | 1)

#define mid (l + r >> 1)

void update(int l, int r, int i, int pos, int val, int c) {

	if (l == r) {

		cnt[i] += c;

		sum[i] += val;

		ans[i] = (cnt[i] > 0 ? sum[i] + lsh[l] : 0);

	} else {

		if (pos <= mid) update(l, mid, ls, pos, val, c);

		else update(mid + 1, r, rs, pos, val, c);

		sum[i] = sum[ls] + sum[rs]; 

		ans[i] = max(ans[ls] + sum[rs], ans[rs]);

	}

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n >> q;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

	}

	for (int i = 1; i <= n; i++) {

		cin >> b[i];

		lsh[++tot] = b[i];

	}

	for (int i = 1; i <= q; i++) {

		cin >> op[i] >> id[i] >> x[i];

		lsh[++tot] = x[i];

	}

	sort(lsh + 1, lsh + tot + 1);

	tot = unique(lsh + 1, lsh + tot + 1) - lsh - 1;

	for (int i = 1; i <= n; i++) {

		b[i] = find(b[i]);

		update(1, tot, 1, b[i], a[i], 1);

	}

	for (int i = 1; i <= q; i++) {

		int pos = id[i];

		x[i] = find(x[i]);

		if (op[i] == 1) {

			update(1, tot, 1, b[pos], lsh[x[i]] - a[pos], 0);

			a[pos] = lsh[x[i]];

		} else {

			update(1, tot, 1, b[pos], -a[pos], -1);

			b[pos] = x[i];

			update(1, tot, 1, b[pos], a[pos], 1);

		}

		cout << ans[1] << '\n';

	}

	return 0;

}
```

## G - 京都观光 / Sightseeing in Kyoto



一个 $n\times m$ 的网格，从 $(1,1)$ 走到 $(n,m)$ ，当你位于 $(x,y)$ 时，可以花费 $a_x$ 走到 $(x,y+1)$ 或花费 $b_y$ 走到 $(x+1,y)$ ，求最小花费。



首先 $n,m\le10^5$ 说明不能使用 DP 了，我们先来一个局部极小值的情况，现在要从 $(x_1,y_1)$ 走到 $(x_2,y_2)$ 只拐一次弯，怎样最小，显然最小值是：



$$\min(a_{x_1}(y_2-y_1)+b_{y_2}(x_2-x_1),a_{x_2}(y_2-y_1)+b_{y_1}(x_2-x_1))$$



即走 $x$ 和 $y$ 方向的两种方案，其中假设第一项比第二项更优，即：



$$a_{x_1}(y_2-y_1)+b_{y_2}(x_2-x_1) \le a_{x_2}(y_2-y_1)+b_{y_1}(x_2-x_1)$$ 



$$(b_{y_2}-b_{y_1})(x_2-x_1) \le (a_{x_2}-a_{x_1})(y_2-y_1)$$



$$\frac{b_{y_1}-b_{y_2}}{y_1-y_2} \le \frac{a_{x_1}-a_{x_2}}{x_1-x_2}$$



变成了一个类似斜率的东西，若先走 $y$ 再走 $x$ 要求 $y$ 方向的斜率更小。



这启发我们在斜率更小的方向贪心地移动，只需维护两个指针然后在 $x$ 和 $y$ 方向走斜率更小的即可。



具体的，应当维护两个方向上的下凸包，对它使用单调栈吧！



```cpp

#include<bits/stdc++.h>

#define int long long

#define X first

#define Y second

using namespace std;

using pii = pair <int, int>;

const int maxn = 1e5 + 5;

int n, m, a[maxn], b[maxn], ans, x = 1, y = 1;

inline double slope(pii p, pii q) {

	return double(p.Y - q.Y) / double(p.X - q.X);

}

struct Hull {

	int top;

	pii st[maxn];

	void insert(pii x) {

		while (top > 1 && slope(st[top - 1], st[top]) > slope(st[top - 1], x)) top--;

		st[++top] = x;

		return;

	}

} A, B;

inline void moveX() {

	ans += B.st[y].Y * (A.st[x + 1].X - A.st[x].X);

	x++;

}

inline void moveY() {

	ans += A.st[x].Y * (B.st[y + 1].X - B.st[y].X);

	y++;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n >> m;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

		A.insert({i, a[i]});

	}

	for (int j = 1; j <= m; j++) {

		cin >> b[j];

		B.insert({j, b[j]});

	}

	while (x < A.top || y < B.top) {

		if (x == A.top) {

			moveY();

		} else if (y == B.top) {

			moveX();

		} else if (slope(A.st[x + 1], A.st[x]) < slope(B.st[y + 1], B.st[y])) {

			moveX();

		} else {

			moveY();

		}

	}

	cout << ans << '\n';

	return 0;

}
```

## J - Mr. Kitayuta vs. Bamboos



有 $n$ 个竹子，第 $i$ 个每天会长高 $h_i$ 米，每天可以选最多 $k$ 个竹子使其下降 $p$ 米，求最后最小的最高竹子高度。



看到最大值最小想到二分，二分最高的竹子高度 $H$ ，于是变成判定问题，所有竹子经过 $m$ 天后是否可以达到 $H$ 米高。正难则反，从最后一天开始倒过来看，每一个竹子都是 $H$ 米，每天会下降 $a_i$ 米，可以拔高其中的 $k$ 个 $p$ 米，很好的模拟，到第一天前若每一颗竹子都不小于 $h_i$ 高，则经过平移 $H$ 是一个合法的答案。



问题在于模拟的过程，不能让竹子在 $m$ 天后低于 $h_i$ 米且过程中不能低于 $0$ 米，则每次贪心地将最快要降为 $0$ 的竹子拔高，用小根堆维护最后一天不能达到 $h_i$ 米的竹子将要将为 $0$ 的天数，按题意倒序模拟即可。



```cpp

#include<bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 1e5 + 5;

	int day, id;

		return a.day > b.day;

	}

	priority_queue <node> q;

	memset(cnt, 0, sizeof cnt);

	for (int i = 1; i <= n; i++) {

		if (x - a[i] * m < h[i]) {

			q.push({x / a[i], i});

		}

	}

	for (int i = 1; i <= m && q.size(); i++) {

		for (int j = 1; j <= k && q.size(); j++) {

			node u = q.top(); q.pop();

			if (u.day < i) return false;

			cnt[u.id]++;

			if (x + cnt[u.id] * p - a[u.id] * m < h[u.id]) {

				q.push({(x + cnt[u.id] * p) / a[u.id], u.id});

			}

		}

	}

	return q.empty();

}
```

## K - BZOJ3252 攻略



给定一颗树，选 $k$ 个根到叶子的路径，求其并集的点权和的最大值。



显然的一个贪心，设根到节点 $u$ 的价值和为 $s_u$ ，先取 $s_u$ 最大的一条链，然后将链上的点权清零并更新 $s$ 重复 $k$ 次即可。怎么求呢，因为有叶子节点这个限制，很难更新，于是想到直接维护所有 $s$ 即可，用树剖 + 数据结构可以做到，但时间复杂度不优，因为总共只有 $n$ 个节点，于是清零时可以一个一个节点清，遇到已经清零过的节点说明其祖先已经被清零了，退出即可，最多清 $n$ 次，每次在数据结构的帮助下只有 $O(\log n)$ 的复杂度。 



具体看线段树，要做到子树减去节点的贡献和求最大值的位置，事实上我们只用维护最大值即可，用线段树上二分求位置。



```cpp

#include<bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 2e5 + 5;

int n, k, w[maxn], u, v, ans, tot, root, vis[maxn], siz[maxn], son[maxn], fat[maxn], sum[maxn], arr[maxn], ide[maxn];

vector <int> G[maxn];

int t[maxn << 2], tag[maxn << 2];

void dfs1(int u, int fa) {

	siz[u] = 1, fat[u] = fa;

	for (auto g : G[u]) {

		if (g == fa) continue;

		dfs1(g, u);

		siz[u] += siz[g];

		if (siz[g] > siz[son[u]]) son[u] = g;

	}

}

void dfs2(int u, int S) {

	ide[u] = ++tot, arr[tot] = u, sum[u] = S + w[u];

	if (son[u]) dfs2(son[u], S + w[u]);

	for (auto g : G[u]) {

		if (g == fat[u] || g == son[u]) continue;

		dfs2(g, S + w[u]);

	}

}

#define ls (i << 1)

#define rs (ls | 1)

#define mid (l + r >> 1) 

void build(int l, int r, int i) {

	if (l == r) {

		t[i] = sum[arr[l]];

	} else {

		build(l, mid, ls);

		build(mid + 1, r, rs);

		t[i] = max(t[ls], t[rs]);

	}

}

void apply(int i, int w) {t[i] += w, tag[i] += w;}

void pushdown(int i) {apply(ls, tag[i]), apply(rs, tag[i]), tag[i] = 0;}

void update(int l, int r, int i, int a, int b, int x) {

	if (a <= l && r <= b) {

		apply(i, x);

	} else {

		pushdown(i);

		if (a <= mid) update(l, mid, ls, a, b, x);

		if (mid < b) update(mid + 1, r, rs, a, b, x);

		t[i] = max(t[ls], t[rs]);

	}

}

pair <int, int> query(int l, int r, int i) {

	if (l == r) {

		return {arr[l], t[i]};

	} else {

		pushdown(i);

		if (t[ls] >= t[rs]) return query(l, mid, ls);

		else return query(mid + 1, r, rs);

	}

}

int faa[maxn];

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n >> k;

	for (int i = 1; i <= n; i++) {

		cin >> w[i];

	}

	for (int i = 1; i < n; i++) {

		cin >> u >> v;

		faa[v] = u;

		G[u].push_back(v);

		G[v].push_back(u);

	}

	for (int i = 1; i <= n; i++) {

		if (!faa[i]) root = i;

	}

	dfs1(root, 0), dfs2(root, 0), build(1, n, 1);

	while (k--) {

		pair <int, int> qu = query(1, n, 1);

		u = qu.first, ans += qu.second;

		while (!vis[u] && u) {

			update(1, n, 1, ide[u], ide[u] + siz[u] - 1, -w[u]);

			w[u] = 0, vis[u] = 1;

			u = fat[u];

		}

	}

	cout << ans << '\n';

	return 0;

}
```

## N - Difficult Mountain



有 $n$ 个人第 $i$ 个人有两个属性 $s_i$ 和 $a_i$ ，现在要攀登一座难度为 $d$ 的张雪峰，当 $d\le s$ 时可以成功攀登并使 $d=\max(d,a_i)$ ，问最优的顺序可以使多少人登山？



将人分为两个集合 $A={i|a_i\le s_i}$ 和 $B={i|s_i<a_i}$ 因为 $A$ 的人可以全部都选，但是 $B$ 中的人选了会导致一些人选不了，所以选 $A$ 更优，必须先选。



看剩下的人，按照 $\max(s_i,a_i)$ 排序，相等则按 $s_i$ 升序排序即可，瞪眼难，证明随便证就行。 



```cpp

#include<bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 5e5 + 5;

int n, d, cnt;

struct node {

	int s, a;

	friend bool operator < (node x, node y) {

		if (max(x.s, x.a) != max(y.s, y.a)) return max(x.s, x.a) < max(y.s, y.a);

		return x.s < y.s;

	}

} p[maxn];

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n >> d;

	for (int i = 1; i <= n; i++) {

		cin >> p[i].s >> p[i].a;

	}

	sort(p + 1, p + n + 1);

	for (int i = 1; i <= n; i++) {

		if (p[i].s >= d) {

			d = max(d, p[i].a);

			cnt++;

		}

	}

	cout << cnt << '\n';

	return 0;

}
```

## O - Modulo Pairing



将长为 $2n$ 的值域为 $[0,m)$ 的序列 $a$ 分为 $n$ 个二元组 $(x,y)$ 求 $(x+y)\mod m$ 的最小值。



首先数据范围太大不考虑 DP 那就是贪心。考虑没有 $m$ 的限制怎么做，显然是第 $i$ 小和第 $i$ 大的数配对，但是加了 $m$ 的限制会导致一些权值变小，但这依然启发我们先将 $a$ 排序。 



依旧 Exchange Argument 解法，将 $a\le b\le c\le d$ 列出来发现 $(a,c)$ $(b,d)$ 的组合一定是不优的，所以数对之间的关系只能是并列或包含。



将数对又分为 $x+y<m$ 和 $x+y\ge m$ 两部分，同样进行分类讨论，一个减去 $m$ 的配对不优，大概可以知道最优配对的样式了：最优配对是分为两个部分，一个部分只有包含关系，即第 $i$ 小加上第 $i$ 大。



分界点越左越好，二分这个分界点即可。 



```cpp

#include<bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 2e5 + 5;

int n, m, a[maxn];

bool check(int pos) {

	for (int i = pos + 1, j = n; i <= j; i++, j--) {

		if (a[i] + a[j] < m) {

			return false;

		}

	}

	return true;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n >> m;

	n <<= 1;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

	}

	sort(a + 1, a + n + 1);

	int l = 0, r = n >> 1, ans;

	while (l <= r) {

		int mid = l + r >> 1;

		if (check(mid << 1)) ans = mid << 1, r = mid - 1;

		else l = mid + 1; 

	}

	int sum = 0;

	for (int i = 1, j = ans; i <= j; i++, j--) sum = max(sum, a[i] + a[j]);

	for (int i = ans + 1, j = n; i <= j; i++, j--) sum = max(sum, a[i] + a[j] - m);

	cout << sum << '\n';

	return 0;
```

## A - The table

首先发现行列的操作可以任意交换顺序，因此不用担心操作先后问题，贪心的想，只要做将会使答案增加的操作即可。



将 $n$ $m$ $\max(a_{i,j})$ 视为同阶，分析时间复杂度，一个优的操作最多使一个元素增加 $O(1)$ 而总共最多增加 $O(n^2)$ ，则最多操作 $O(n^2)$ 次，一次操作检查每个行/列一次，时间复杂度为 $O(n^4)$

## B - Catshock



有 $n$ 个节点的树，猫在节点 $1$ 而人在节点 $n$ 。有两种操作，操作 $1$ 让猫随机到达相邻节点，操作 $2$ 删除一个节点和它连的边，寻找一个操作方案使得猫到达 $n$ ，不能连续两个操作 $2$ ，答案长度不超过 $3n$ 。



猫会乱走，应当限制。看到 $3n$ 启发我们让猫走两步在删除一个点，此时猫的限制是猫到 $1$ 的距离一定为偶数，所以只要删除到 $1$ 距离为奇数的叶子节点既可以限制猫，又不会删除猫或使猫与人不在一个连通块。那没有偶数距离的叶子节点了怎么办，这时让猫再走一步，同理删除奇数距离叶子即可。最后树变为一奇一偶，判断猫是否在 $n$ 即可。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 2e5 + 10;

int T, n, u, v, vis[maxn], dis[maxn], deg[maxn];

vector <int> G[maxn], path;

bool dfs(int u, int fa) {

	if (u == n) {

		path.push_back(u);

		vis[u] = true;

	}

	for (auto v : G[u]) {

		if (v == fa) continue;

		dis[v] = dis[u] ^ 1;

		if (dfs(v, u)) {

			path.push_back(u);

			vis[u] = true;

		}

	}

	return vis[u];

}

void solve() {

	cin >> n;

	path.clear();

	for (int i = 1; i <= n; i++) {

		G[i].clear();

		dis[i] = deg[i] = 0;

		vis[i] = false;

	}

	for (int i = 1; i < n; i++) {

		cin >> u >> v;

		deg[u]++, deg[v]++;

		G[u].push_back(v);

		G[v].push_back(u);

	}

	dfs(1, 0);

	reverse(path.begin(), path.end());

	queue <int> que;

	for (int i = 1; i <= n; i++) if (deg[i] == 1 && !vis[i]) que.push(i), vis[i] = true;

	int step = 0;

	vector <pair <int, int>> ans;

	while (!que.empty()) {

		int u = que.front(); que.pop();

		ans.push_back({1, 0}), step ^= 1;

		if (step == dis[u]) ans.push_back({1, 0}), step ^= 1;

		ans.push_back({2, u});

		for (auto g : G[u]) {

			if (vis[g]) continue; deg[g]--;

			if (deg[g] == 1) vis[g] = true, que.push(g);

		}

	}

	for (int i = 0; i < path.size() - 1; i++) {

		ans.push_back({1, 0});

		step ^= 1;

		if (dis[path[i]] == step) {

			ans.push_back({1, 0});

			step ^= 1;

		}

		ans.push_back({2, path[i]});

	}

	cout << ans.size() << '\n';

	for (auto S : ans) {

		if (S.first == 1) cout << S.first << '\n';

		else cout << S.first << " " << S.second << '\n';

	}

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> T;

	while (T--) solve();

	return 0;

}

```



## C - Prime Problem



将 $n$ 个数分为尽量小的集合，使得每个集合元素之和为素数。



发现哥德巴赫猜想至少在 OI 的数据范围内一定成立，即一个大于 $2$ 的偶数一定可以拆分为两个素数之和，于是设 $S=\sum_{i=1}^ni$ 然后大力分类讨论。



$S$ 是素数时，整体为一个集合。 



$S$ 是偶数时，将其拆分为两个素数，可以发现较小的 $p$ 不大于 $n$ ，此时将其分为 ${p}$ 和 ${p}$ 的补集。



$S$ 是奇数时，若可以拆分为 $2$ 和一个奇素数之和，则分为 ${2}$ 与其补集；否则拆分为 $3$ 和一个偶数之和，然后像情况二一样做即可。 



```cpp

#include<bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 6005;

bool check(int x) {

	for (int i = 2; i * i <= x; i++) {

		if (x % i == 0) return false;

	}

	return true;

}

int find(int x) {

	for (int i = 2; i <= n; i++) {

		if (check(i) && check(x - i)) {

			return i;

		}

	}

	return 0;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n;

	S = n * (n + 1) / 2;

	if (check(S)) {

		for (int i = 1; i <= n; i++) cout << 1 << ' ';

	} else if (S & 1) {

		if (check(S - 2)) {

			for (int i = 1; i <= n; i++) cout << (i == 2 ? 2 : 1) << ' ';

			return 0;

		}

		int x = find(S - 3);

		for (int i = 1; i <= n; i++) {

			if (i == 3) cout << 3 << ' ';

			else if (i == x) cout << 2 << ' ';

			else cout << 1 << ' ';

		}

	} else {

		int x = find(S);

		for (int i = 1; i <= n; i++) cout << (i == x ? 2 : 1) << ' ';

	}

	return 0;

}
```

## D - Prefix Product Sequence



给定 $n$ 求一个 $[1,n]$ 的排列 $a$ 使其模 $n$ 意义下的前缀乘积为一个 $[0,n)$ 的排列。



事实上在模 $n$ 意义下 $[1,n]$ 的排列等价于 $[0,n)$ 的排列。设前缀乘积为 $s$ 则不存在不同的整数 $i$ $j$ 使得 $s_i=s_j$ ，即 $a$ 不存在任意一个子段使得其乘积为 $1$ 或 $0$ ，那么显然 $1$ 要放在最开头， $n$ 放在结尾。



考虑序列中有一段是 $(n-1)! \mod n \neq 1$ ，这个条件当且仅当 $n$ 是大于 $4$ 的合数无解，即 $n$ 为 $1$ 或 $2$ 或 $3$ 或 $4$ 或更大的素数。考虑 $n$ 为素数怎么做，只要放入 $\frac{i1}{i-1}$ 就能做到没有重复。



```cpp

#include<bits/stdc++.h>

		a = a * a % n, b >>= 1;

	} else {

		if (check(n)) {

			cout << "YES\n";

			for (int i = 1; i <= n; i++) {

				if (i == 1 || i == n) {

					cout << i << '\n';

				} else {
```

## E - Double Knapsack



给定两个集合 $A$ $B$ 求出两个集合元素之和相等的子集（值域为 $[1,n]$ ）。



很妙的做法，设两个子集之和的差值为 $x$ 若，哪边小了就随便加一个进去，若有解那么做 $2n$ 次一定可以得到，因为差值的值有 $2n$ 个而总共有 $2n+1$ 个差值，则一定存在至少两个重复的差值，那么做差就是答案。 



```cpp

#include <bits/stdc++.h>

#define int long long

const int maxn = 2e6 + 5;

	for (int i = 1; i <= n; i++) cin >> a[i];

	for (int i = 1; i <= n; i++) cin >> b[i];

	int T = n << 1;

	mp[0] = {0, 0};

	while (T--) {

		if (x < 0) x += a[++p1];

		else x -= b[++p2];

		if (mp.count(x)) {

			int s1 = mp[x].first, s2 = mp[x].second;

			cout << p1 - s1 << '\n';

			for (int i = s1 + 1; i <= p1; i++) cout << i << ' '; cout << '\n';

			cout << p2 - s2 << '\n';

			for (int i = s2 + 1; i <= p2; i++) cout << i << ' '; cout << '\n';

			return 0;

		}

		mp[x] = {p1, p2};

	}

	cout << -1 << '\n';

	return 0;

}
```

## F - Square Root of Permutation



有两个 $[1,n]$ 的排列 $q$ 和 $p$ 满足 $q_{q_i}=p_i$ 已知 $p$ 求 $q$ 。



将 $i$ 向 $p_i$ 连边，那么整个图应为若干个环，尝试将一个环做这样的操作，发现若是奇环，则环不变，否则会变成两个大小相等的环，其相邻元素不在同一个环。对于 $p$ 只要看其中的偶环按照大小两两配对是否可行，奇环 



具体的，奇环变为 ```1 3 5 ... k 2 4 6 ... k-1``` 偶环变为 ```1 3 5 ... k-1``` 和 ```2 4 6 ... k``` 。 



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 1e6 + 5; 

int n, tot, p[maxn], q[maxn], vis[maxn];

vector <int> G[maxn], len[maxn];

bitset <maxn> bit;

void solve1(vector <int> vec) {

	int len = vec.size();

	vector <int> tmp;

	tmp.push_back(vec[0]);

	for (int i = 1, j = ((len - 1) >> 1) + 1; j < len; i++, j++)

		tmp.push_back(vec[j]), tmp.push_back(vec[i]);

	q[tmp[len - 1]]=tmp[0];

	for (int i = 1; i < len; i++) q[tmp[i - 1]] = tmp[i];

}

void solve2(vector <int> v1, vector <int> v2) {

	int len = v1.size();

	vector <int> tmp;

	for (int i = 0; i < len; i++) tmp.push_back(v1[i]), tmp.push_back(v2[i]);

	q[tmp[(len << 1) - 1]] = tmp[0];

	for (int i = 1; i < (len << 1); i++) q[tmp[i - 1]] = tmp[i];

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> p[i];

	}

	for (int i = 1; i <= n; i++) {

		if (vis[i]) continue;

		int u = i; tot++;

		while (!vis[u]) {

			G[tot].push_back(u);

			vis[u] = 1, u = p[u];

		}

		len[G[tot].size()].push_back(tot);

	}

	for (int i = 1; i <= n; i++) {

		if (!len[i].size()) continue;

		if (!(i & 1) && (len[i].size() & 1)) {

			cout << "-1\n";

			return 0;

		}

		if (i & 1) {

			for (auto j : len[i]) {

				solve1(G[j]);

			}

		} else {

			for (int j = 0; j < len[i].size(); j += 2) {

				solve2(G[len[i][j]], G[len[i][j + 1]]);

			}

		}

	}

	for (int i = 1; i <= n; i++) {

		cout << q[i] << ' ';

	}

	return 0;

}
```

## G - Hack it!



定义 $f(x)$ 为十进制下 $x$ 个各数位之和，给定模数 $a$ 求满足 $\sum_{i=l}^{r}f(i)\equiv 0 \mod a$ 的 $[,l r]$ ，答案在 $10^{200}$ 内。



设 $p \equiv \sum_{i=0}^{10^{18}-1}f(i)$ ，设 $g(x) = \sum_{i=x}^{10^{18}+x-1}f(i)$ 易得 $g(x)\equiv p+x$ 则可以构造一组解为 $[a-p,a-p+10^{18}-1]$ 。



现在来求 $p$ 。每个数位上的数都平均出现，有 $18$ 个数位，数之和为 $45$ 排列组合后得 $p=18\times45\times10^{17}=81\times10^{18}\mod a$ 。 



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

int a, p, inf = 1e18;

int ksm(int x, int b) {

	int res = 0;

	while (b) {

		if (b & 1) (res += x) %= a;

		(x += x) %= a, b >>= 1;

	}

	return res;

}

signed main() {

	ios::sync_with_stdio(0);
```

## H - Bitwise Reversion



给定三个数 $x$ $y$ $z$ 问是否有三个数 $a$ $b$ $c$ 使得 $a&b=x$ $b&c=y$ &a&c=z& 。



唐题，显然每个位的答案是没有关联的，一个一个位处理即可，现在假设这 $(x,y,z)$ 第 $i$ 位的集合为 $S_i$ ， $(a,b,c)$ 的第 $i$ 位的集合为 $A_i$ 。

```cpp
	}
```

## I - Symmetrical Polygons



给定 $n$ 条边，选取其中的一些组成一个轴对称的非退化凸多边形，求其最大边长。



首先将相同长度的边两两一对贪心加入，剩下来是没有组对的边，可以选 $0$ 或 $1$ 或 $2$ 条，类比等腰三角形满足边小于两腰之和，要周长最大，只需去最大和满足的次大即可。 



```cpp

#include<bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 2e5 + 5;

int T, n, a[maxn];

map <int, int> mp;

void solve() {

	mp.clear();

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

		mp[a[i]]++;

	}

	queue <int> q1;

	priority_queue<int, vector<int>, greater <int>> q2;

	int sum = 0;

	for (int i = 1; i <= n; i++) {

		int k = mp[a[i]];

		if (k > 1) {

			for (int j = 1; j <= (k >> 1); j++) {

				mp[a[i]] -= 2;

				q1.push(a[i]);

				sum += a[i];

			}

		}	

		if (mp[a[i]]) q2.push(a[i]);	

		mp[a[i]] = 0;

	}	

	int lst = 0, ans = 0;

	if (q1.size() != 1) ans = sum << 1;

	while (q2.size()) {

		int x = q2.top(); q2.pop();

		if ((sum << 1) + lst > x) ans = max(ans, (sum << 1) + lst + x);

		lst = x;

	}

	cout << ans << '\n';

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> T;

	while (T--) solve();

	return 0;

}
```

## J - Not Alone



给定长为 $n$ 的环形序列，一次操作可以使一个元素 $+1$ 或 $-1$ ，至少多少次操作可以使序列任意一项等于相邻一项。



首先想到破环为链，把相等的当作一个联通块，很容易想到极小连通块大小是 $2$ 个或 $3$ 个，先在这个基础上做 DP 然后修缮环处的答案。 



分类讨论开头结尾的情况，不多，一种情况做一次 DP 最后取最小值。 



```cpp

#include <bits/stdc++.h>

using namespace std;

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

		if (i <= 4) a[n + i] = a[i];

	}

	ans = 1e18;

	for (int i = 0; i <= 3; i++) {

		for (int j = 0; j <= n + i; j++) f[j] = 1e18;

		}

		ans = min(ans, f[n + i]);

	}

	cout << ans << '\n';

}
```

## B - 最长同类子串



给定两个字符串 $s$ 和 $t$ ，求最大长度 $k$ 使得两个字符串中存在长为 $k$ 的子串 $s'$ 和 $t'$ 使其本质相同，即若 $s'_i=s'_j$ 则 $t'_i=t'_j$ 。



先想到二分 $k$ 然后将问题转化为判定问题，设计一种多项式哈希，第 $i$ 个数的权值为其到前一个相同元素的距离，若没有则为 $0$ ，这不能使用前缀和求子串哈希，只能用滑动窗口动态维护。 



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

using ull = unsigned long long;

const int maxn = 1e5 + 5, base = 998244353;

int n, m, preS[maxn], preT[maxn], nxtS[maxn], nxtT[maxn];

ull pw[maxn] = {1};

char s[maxn], t[maxn];

void build() {

	for (int i = 1; i < maxn; i++) pw[i] = pw[i - 1] * base;

	auto make1 = [&](char *str, int len, int *pre) -> void{

		vector <int> lst(26, 0);

		for (int i = 1; i <= len; i++) {

			pre[i] = lst[str[i] - 'a'];

			lst[str[i] - 'a'] = i;

		}

		return;

	};

	auto make2 = [&](char *str, int len, int *pre) -> void{

		vector <int> lst(26, len + 1);

		for (int i = len; i >= 1; i--) {

			pre[i] = lst[str[i] - 'a'];

			lst[str[i] - 'a'] = i;

		}

		return;

	};

	make1(s, n, preS), make1(t, m, preT);

	make2(s, n, nxtS), make2(t, m, nxtT);

	return;

}

bool check(int k) {

	auto calc = [&](char *str, int len, int *pre, int *nxt) -> vector <ull>{

		vector <ull> ret, w(len + 1); ull now = 0;

		for (int i = 1; i <= k; i++) {

			w[i] = (pre[i] <= i - k || !pre[i] ? 0 : i - pre[i]);

			(now *= base) += w[i];

		}

		ret.push_back(now);

		for (int i = k + 1; i <= len; i++) {

			w[i] = (pre[i] <= i - k || !pre[i] ? 0 : i - pre[i]);

			now = now * base - w[i - k] * pw[k];

			if (nxt[i - k] <= i) {

				now -= w[nxt[i - k]] * pw[i - nxt[i - k]];

				w[nxt[i - k]] = 0;

			}

			now += w[i];

			ret.push_back(now);

		}

		return ret;

	};

	vector <ull> hshS = calc(s, n, preS, nxtS), hshT = calc(t, m, preT, nxtT);

	unordered_map <ull, bool> mp;

	for (auto tmp : hshS) mp[tmp] = true;

	for (auto tmp : hshT) if (mp[tmp]) return true;

	return false;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> (s + 1) >> (t + 1);

	n = strlen(s + 1), m = strlen(t + 1);

	build();

	int l = 1, r = min(n, m), ans = 0;

	while (l <= r) {

		int mid = l + r >> 1;

		if (check(mid)) l = mid + 1, ans = mid;

		else r = mid - 1;

	}

	cout << ans << '\n'; 

	return 0;

}

```



## C - Prefix Function Queries



KMP 自动机是弱化版的 AC 自动机，相当于单串的 AC 自动机。先建一个 fail 树，设 $nxt_{i,j}$ 表示从 $i$ 开始沿着 fail 树向上跳得到的最大的位置 $k$ 使 $s_{k+1}=j$ 。



注意到此时我们可以把 KMP 求的过程表示为：对于 $i$ ，从 $fail_{i−1}$ 开始沿着 fail 树向上跳找到的最大的位置 $k$ 使 $s_{k+1}=s_i$ ，若不存在则为 $0$ 。这也可以让我们得到 $fail_i=nxt_{fail_{i−1,s_i}}$ 。



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 1e6 + 15;

int n, m, q, fail[maxn], nxt[maxn][26];

char s[maxn];

void build(){

	int n = strlen(&s[1]);

	for (int i = 2, j = 0; i <= n; i++) {

		while (j >= 1 && s[i] != s[j + 1]) j = fail[j];

		if (s[i] == s[j + 1]) j++;

		fail[i] = j;

	}

	for (int i = 1; i <= n; i++) {

		for (int j = 0; j < 26; j++) {

			nxt[i - 1][j] = nxt[fail[i - 1]][j];

		}

		nxt[i - 1][s[i] - 'a'] = i - 1;

	}

	return;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> &s[1] >> q;

	n = strlen(&s[1]);

	build();

	for (int i = 1; i <= q; i++) {

		cin >> &s[n + 1];

		m = n + strlen(&s[n + 1]);

		for (int j = n + 1, k = fail[n]; j <= m; j++) {

			k = nxt[k][s[j] - 'a'];

			if (s[j] == s[k + 1]) k++;

			fail[j] = k;

			cout << k << ' ';

			for (int c = 0; c < 26; c++) {

				nxt[j - 1][c] = nxt[fail[j - 1]][c];

			}

			nxt[j - 1][s[j] - 'a'] = j - 1;

		}

		cout << '\n';

	}

	return 0;

}
```

## E - 失配树

首先跑一遍 KMP 得到 $nxt$ 数组，则长为 $i$ 的前缀的 border 长度为 $nxt_i,nxt_{nxt_i}...$ 。考虑以 $i$ 到 $nxt_i$ 建树，这就是 AC 自动机中只有一个字符串的 fail 树，在树上跑 LCA 即为答案。值得注意的是一个子串的 border 一定小于子串长度。

## F - 葫芦和斌斌的字符串2



给定字符串 $s$ 有两种操作：



1. 在字符串后添加一个字符。

2. 找到字符串最长的出现 $k$ 次的 border 。



给定 $s$ 和 $q$ 次操作，操作一是在 $s$ 尾部加入一个新字符，操作二是求 $s$ 的最长满足在 $s$ 中出现至少 $k$ 次的 border 。



首先建 border 树，发现从 $n$ 往上跳，跳到第一个子树大小 $\ge k$ 的节点即为答案。动态维护很棘手，于是离线下来后树剖+树状数组搞过去。 



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 2e6 + 5;

int n, q, len, op[maxn], k[maxn], N[maxn], dep[maxn], nxt[maxn][20], t[maxn], siz[maxn], son[maxn], top[maxn], ide[maxn], tot;

vector <int> G[maxn];

char s[maxn];

void insert(int i) {

	while (len && s[i] != s[len + 1]) len = nxt[len][0];

	if (s[len + 1] == s[i] && i != 1) len++;

	nxt[i][0] = len, G[len].push_back(i);

	for (int j = 1; j < 20; j++) {

		nxt[i][j] = nxt[nxt[i][j - 1]][j - 1];

	}

}

void makedep(int i) {

	int u = i;

	for (int j = 19; j >= 0; j--) {

		if (nxt[u][j]) {

			u = nxt[u][j];

			dep[i] += (1 << j);

		}

	}

	return;

}

void add(int x, int y) { for (; x <= N[q]; x += x & -x) t[x] += y; }

int sum(int x, int y = 0) { for (; x; x -= x & -x) y += t[x]; return y; }

int upd(int l, int r, int x) { add(l, x), add(r + 1, -x); }

void dfs1(int u, int fa) {

	siz[u] = 1;

	for (auto g : G[u]) {

		dfs1(g, u);

		siz[u] += siz[g];

		if (siz[g] > siz[son[u]]) son[u] = g;

	}

}

void dfs2(int u, int T) {

	top[u] = T, ide[u] = ++tot;

	if (son[u]) dfs2(son[u], T);

	for (auto g : G[u]) {

		if (g == son[u]) continue;

		dfs2(g, g);

	}

}

int dfs3(int u, int fa) {

	if (u > n) return 0;

	int sz = 1;

	for (auto g : G[u]) {

		sz += dfs3(g, u);

	}

	upd(ide[u], ide[u], sz);

	return sz;

}

void update(int u) {

	while (top[u]) {

		upd(ide[top[u]], ide[u], 1);

		u = nxt[top[u]][0];

	}

	upd(ide[top[u]], ide[u], 1);

}

int query(int u, int x) {

	for (int i = 19; i >= 0; i--) {

		if (sum(ide[nxt[u][i]]) < x) {

			u = nxt[u][i];

		}

	}

	return nxt[u][0];

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n >> q >> (s + 1); N[0] = n;

	for (int i = 1; i <= q; i++) {

		cin >> op[i]; N[i] = N[i - 1];

		if (op[i] == 1) {

			cin >> s[++N[i]];

		} else {

			cin >> k[i];

		}

	} 

	

	for (int i = 1; i <= N[q]; i++) insert(i);

	for (int i = 1; i <= N[q]; i++) makedep(i);

	dfs1(0, -1), dfs2(0, 0), dfs3(0, -1);

	for (int i = 1; i <= q; i++) {

		if (op[i] == 1) {

			update(N[i]);

		} else {

			int ans = query(N[i], k[i]);

			if (ans) cout << ans << '\n';

			else cout << "-1\n";

		}

	}

	return 0;

}
```

## H - SZA-Template



问 $s$ 可以被一个子串重叠组成，求该子串长度最大值。



先做一遍 kmp 是肯定的，因为答案一定是 $s$ 的 border 。设 $f_i$ 为前 $i$ 个的答案，若 $f_i=f_{border_i}$ 当且仅当存在 $j$ 使得 $f_j=f_{border_i}$ 且 $i-border_i\le j$ ，否则 $f_i=i$ 。 



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 5e5 + 5;

int n, f[maxn], b[maxn], nxt[maxn];

char s[maxn];

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> (s + 1);

	n = strlen(s + 1);

	for (int i = 2, len = 0; i <= n; i++) {

		while (len && s[len + 1] != s[i]) len = nxt[len];

		if (s[len + 1] == s[i]) len++;

		nxt[i] = len;

	}

	f[1] = b[1] = 1;

	for (int i = 2; i <= n; i++) {

		f[i] = i;

		if (b[f[nxt[i]]] >= i - nxt[i]) f[i] = f[nxt[i]];

		b[f[i]] = i;

	}

	cout << f[n] << '\n';

	return 0;

}
```

## J - PRE-Prefixuffix



选取 $s$ 的长为 $L$ 的前后缀，求使前后缀循环同构的最大 $L$ 。



循环同构的前后缀使得字符串为 $ABSBA$ 的形式，设 $s_{1...n}$ 的 border 为 $A$ ，而 $s_{|A|+1,...,n-|A|}$ 的 border 为 $B$ ，答案为 $\max(|A|+|B|)$ 。



首先 $A$ 是好求的，用一次 KMP 就可以将所有 $A$ 求出。求 $B$ 时从中心扩展将所有 $B$ 求出，设 $[i+1,n-i]$ 的 border 长度为 $L$ 则 $[i,n-i+1]$ 的 border 长度不大于直接转移的 $L+2$ ，若不再是 border 则减小 $L$ 判断，总共最多只有 $L$ 次减少。 



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

using ull = unsigned long long;

const int maxn = 1e6 + 5, base = 131, mod = 1e9 + 7;

int n, nxt[maxn], bor[maxn], ans;

ull h[maxn], pw[maxn] = {1};

char s[maxn];

inline int Hash(int l, int r) {

	return (h[r] - h[l - 1] * pw[r - l + 1] % mod + mod) % mod;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	cin >> n >> (s + 1);

	for (int i = 1; i <= n; i++) {

		pw[i] = pw[i - 1] * base % mod;

		h[i] = (h[i - 1] * base % mod + s[i]) % mod;

	}

	for (int i = 2, len = 0; i <= n; i++) {

		while (len && s[len + 1] != s[i]) len = nxt[len];

		if (s[len + 1] == s[i]) len++;

		nxt[i] = len;

	}

	for (int i = n >> 1; i; i--) {

		bor[i] = bor[i + 1] + 2;

		while (bor[i]) {

			if ((i + bor[i]) << 1 > n) bor[i]--;

			else if (Hash(i + 1, i + bor[i]) != Hash(n - i - bor[i] + 1, n - i)) bor[i]--;

			else break;

		}

	}

	for (int i = nxt[n]; i; i = nxt[i]) if ((i << 1) <= n) ans = max(ans, bor[i] + i);

	cout << ans << '\n';

	return 0;

}

```
