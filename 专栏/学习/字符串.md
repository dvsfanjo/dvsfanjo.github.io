# 字符串哈希



字符串哈希的一般方式是将其变为一个 base 进制数并对一个大整数取模，如下：



```cpp

const int base = 1e9 + 7; //base要是质数，越sb的质数越难卡

ull H; //使用 unsigned long long 的自然溢出取模会很方便

char s[maxn];

for (int i = 1; i <= n; i++) {

    H = H * base + s[i];

}

```



利用前缀和的思想，我们可以求字符串子串的哈希值。



```cpp

ull ha[maxn], pw[maxn] = {1};

cin >> (s + 1);

n = strlen(s + 1);

for (int i = 1; i <= n; i++) {

    pw[i] = pw[i - 1] * base;

    ha[i] = ha[i - 1] * base + s[i];

}

```



使用这个 Hash(l, r) 可以得到子串 $s[l...r]$ 的哈希值，两个子串相等时，他们的哈希值是一样的。



```cpp

ull Hash(int l, int r) {

    return ha[r] - ha[l - 1] * pw[r - l + 1];

}

```



## 例题



### [UVA10298 Power Strings](https://www.luogu.com.cn/problem/UVA10298)



:::info[题意]

给定一个字符串，求一个字符串最多由多少个重复的子串连接而成。

:::

:::success[思路+代码]

枚举长度的因数，逐个判断子串即可。



时间复杂度 $O(\sqrt{n}+\sum_{d|n}\frac{n}{d})=O(n)$



```cpp

#include <bits/stdc++.h>

using namespace std;

using ull = unsigned long long;

const int maxn = 1e6 + 5, base = 1e9 + 7;

int n;

ull ha[maxn], pw[maxn] = {1};

char s[maxn];

ull Hash(int l, int r) {

	return ha[r] - ha[l - 1] * pw[r - l + 1];

}

bool check(int len) {

	ull H = Hash(1, len);

	for (int i = len + 1; i <= n; i += len) {

		if (H != Hash(i, i + len - 1)) {

			return false;

		}

	}

	return true;

}

void solve() {

	n = strlen(s + 1);

	if (n == 1 && s[1] == '.') exit(0);

	for (int i = 1; i <= n; i++) {

		ha[i] = ha[i - 1] * base + s[i];

	}

	int ans = 1;

	for (int d = 1; d * d <= n; d++) {

		if (n % d) continue;

		if (check(d)) ans = max(ans, n / d);

		if (check(n / d)) ans = max(ans, d);

	}

	cout << ans << '\n';

	return;

}

int main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	for (int i = 1; i < maxn; i++) {

		pw[i] = pw[i - 1] * base;

	} 

	while (cin >> (s + 1)) {

		solve();

	}

	return 0;

}

```

### [P2757 [国家集训队] 等差子序列](https://www.luogu.com.cn/problem/P2757)



:::info[题意]

给定一个 $1$ ~ $n$ 的排列，判断是否存在一个等差子序列

:::

:::success[思路+代码]

等差子序列长度最小为 3 故我们只用判断有没有先后出现 $a-d$ 、 $a$ 和 $a+d$ 即可。



从左到右扫描线，对于 $a$ 如果先前出现了 $a-d$ 却没有出现 $a+d$ ，那么一定会有一个等差子序列。



换而言之，问题转化为了值域上的动态判断回文。



我们只用正、反分别维护哈希，就可以判断回文，又因为是动态的，我们用线段树维护哈希即可。



```cpp

#include <bits/stdc++.h>

using namespace std;

using ll = long long;

using ull = unsigned long long;

const int maxn = 5e5 + 5, base = 13331, p = 1e9 + 7;

ll T, n, a[maxn], t[maxn << 2][2], pw[maxn];

#define ls (i << 1)

#define rs (ls | 1)

#define mid (l + r >> 1)

void pushup(ll l, ll r, ll i) {

	t[i][0] = (t[ls][0] * pw[r - mid] % p + t[rs][0]) % p;

	t[i][1] = (t[rs][1] * pw[mid - l + 1] % p + t[ls][1]) % p;

	return;

}

void build(ll l, ll r, ll i) {

	if (l == r) {

		t[i][0] = t[i][1] = 0;

	} else {

		build(l, mid, ls);

		build(mid + 1, r, rs);

		pushup(l, r, i);

	}

	return;

}

void update(ll l, ll r, ll i, ll pos) {

	if (l == r) {

		t[i][0] = t[i][1] = 1;

	} else {

		if (pos <= mid) update(l, mid, ls, pos);

		else update(mid + 1, r, rs, pos);

		pushup(l, r, i);

	}

	return;

}

ll res1, res2, num;

void query1(ll l, ll r, ll i, ll a, ll b) {

	if (a <= l && r <= b) {

		res1 = (res1 + t[i][0] * pw[num] % p) % p;

		num += r - l + 1;

	} else {

		if (mid < b) query1(mid + 1, r, rs, a, b);

		if (a <= mid) query1(l, mid, ls, a, b);

	}

	return;

}

void query2(ll l, ll r, ll i, ll a, ll b) {

	if (a <= l && r <= b) {

		res2 = (res2 + t[i][1] * pw[num] % p) % p;

		num += r - l + 1;

	} else {

		if (a <= mid) query2(l, mid, ls, a, b);

		if (mid < b) query2(mid + 1, r, rs, a, b);

	}

	return;

}

bool solve() {

	cin >> n;

	for (int i = 1; i <= n; i++) {

		cin >> a[i];

	}

	build(1, n, 1);

	for (int i = 1; i <= n; i++) {

		update(1, n, 1, a[i]);

		ll len = min(a[i], n - a[i] + 1);

		res1 = res2 = 0;

		num = 0; query1(1, n, 1, a[i] - len + 1, a[i] + len - 1);

		num = 0; query2(1, n, 1, a[i] - len + 1, a[i] + len - 1);

		if (res1 != res2) return true;

	}

	return false;

}

int main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	pw[0] = 1;

	for (int i = 1; i < maxn; i++) {

		pw[i] = pw[i - 1] * base % p;

	}

	cin >> T;

	while (T--) {

		cout << (solve() ? "Y\n" : "N\n");

	}

	return 0;

} 

```

:::



使用字符串哈希+二分可以求 border 和最长回文半径，当然这是 $O(n\log n)$ 的远劣与 $O(n)$ 的 KMP 和 manacher 。



# KMP



KMP 的核心在于计算每一个前缀最长 border 的方法（这里 border 是指子串 $S[1...i]$ 中同时是真前缀和真后缀的子串长度）。



用 $nxt(i)$ 表示前缀 $S[1...i]$ 的 border 长度，如果已知 $nxt(1...i-1)$ 如何求 $nxt(i)$ 呢。



1. 如果 $s[nxt(i-1)+1]=s[i]$ 即 $nxt(i-1)$ 向后扩展一个下标后是 $nxt(i)$ ，可以得到 $nxt(i)=nxt(i-1)+1$ 。

2. 不能向后扩展，我们只能尝试更小的 border 进行匹配，即 $nxt(nxt(i-1))$ ，若依旧不行就尝试 $nxt(nxt(nxt(i-1)))$ 直到成功。



核心代码：



```cpp

int len = 0;

for (int i = 2; i <= n; i++) { //下标从 2 开始是因为很显然 nxt[1]=0

    while (len && s[len + 1] != s[i]) len = nxt[len];

    if (s[len + 1] == s[i]) len++;

    nxt[i] = len;

}

```



那么如何进行模式串匹配呢？



```

模式串：abcad

文本串：abcacabcada

```



这时我们匹配到第 4 位后发现匹配不了了，考虑暴力，我们会将模式串从文本串第 2 位开始继续匹配，但这时间复杂度依旧很劣，但是我们可以从第 4 位重新开始匹配，因为文本串前缀 $s[1...4]$ 的 border 为 $1$ 。



```

模式串：   abcad

文本串：abcacabcada

```



失配了，我们从第 5 位开始，因为 $nxt(1)=0$ 。



```

模式串：    abcad

文本串：abcacabcada

```



依旧失配，进一位。



```

模式串：     abcad

文本串：abcacabcada

```



成功匹配！因此我们模式串匹配的核心代码长这样：



```cpp

int len = 0;

for (int i = 1; i <= n; i++) {

    while (len > 0 && s[i] != t[len + 1]) len = nxt[len];

    if (s[i] == t[len + 1]) len++;

    if (len ==  m) {

        //匹配成功，说明 s[i-m+1...i]=t

        len = nxt[len];

    }

}

```



注意到，变量 len 只会在匹配到时增加 $1$ ，而 len 是非负数，减少的次数不会多余增加的次数，增加最多 $n-1$ 次，故 KMP 算法时间复杂度应为 $O(n)$ 。

### [P4391 [BalticOI 2009] Radio Transmission 无线传输](https://www.luogu.com.cn/problem/P4391)



:::info[题意]

给你一个字符串 $s_1$，它是由某个字符串 $s_2$ 不断自我连接形成的（保证至少重复 $2$ 次）。但是字符串 $s_2$ 是不确定的，现在只想知道它的最短长度是多少。

:::



:::success[思路+代码]

假设最短长度为 $x$ ，那么 $nxt(1...x)$ 都为 $0$ ， $nxt(x+1)=1$ ， $nxt(x+2)=2$ 等等。那么 $x = n - nxt(n)$ 。 

```cpp

#include <bits/stdc++.h>

using namespace std;

const int maxn = 2e6 + 5;

int n, nxt[maxn];

char a[maxn];

int main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n >> (a + 1);

	int len = 0;

	for (int i = 2; i <= n; i++) {

		while (len && a[len + 1] != a[i]) len = nxt[len];

		if (a[len + 1] == a[i]) len++;

		nxt[i] = len;

	}

	cout << n - nxt[n] << '\n';

	return 0;

}
```

### [UVA1328 Period](https://www.luogu.com.cn/problem/UVA1328)

:::info[题意]

求字符串每一个前缀是否是周期串，若是，输出前缀长度和循环节数量。

:::

:::success[思路+代码]

border 有一个性质：若 border 长度 $k$ 大于字符串的一半，且 $n$ 能被 $n-k$ 整除，那么字符串就是周期串，且 $n-k$ 就是字符串最长循环节的长度。

```cpp

#include <bits/stdc++.h>

using namespace std;

using ll = long long;

const int maxn = 2e6 + 5;

int n, nxt[maxn], T;

char a[maxn];

void solve() {

	cin >> (a + 1);

	int len = 0;

	cout << "Test case #" << ++T << '\n';

	for (int i = 2; i <= n; i++) {

		while (len && a[len + 1] != a[i]) len = nxt[len];

		if (a[len + 1] == a[i]) len++;

		nxt[i] = len;

		if (i % (i - nxt[i]) == 0 && i / (i - nxt[i]) > 1) {

			cout << i << ' ' << i / (i - nxt[i]) << '\n';

		}

	}

	cout << '\n';

	return;

} 

int main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	while (cin >> n && n) {

		solve();

	}

	return 0;

}

```

# exKMP Z函数



exKMP 要求的东西和 KMP 有点像（当然也是有差别的），我们定义 $z(i)$ 为 $s$ 与以 $i$ 开头的 $s$ 的后缀的最长公共前缀的长度，即 LCP 的长度（其实就是有不同限制条件的 border 啦）。



我们将维护一个右端点最大的匹配区间 $[l,r]$ 和 $z(1...i-1)$ 来求出 $z(i)$ ，我们分为以下几种情况：



- $i\le r$ 时，有 $s[i-l,r-l]=s[i,r]$ 故 $z(i)\ge \min(z(i-l),r-i+1)$ ，若 $z(i-l)\le r-i+1$ 则 $z(i)=z(i-l)$ 否则令 $z(i)=r-i+1$ 再暴力扩展

- $l\ge r$ 时，直接暴力扩展



最后用 $z(i)$ 更新 $[l,r]$ 即可。



```cpp

void Z(char *s, int n) {

	for (int i = 1; i <= n; i++) z[i] = i == 1 ? n : 0; //下标从 1 开始

	for (int i = 2, l = 0, r = 0; i <= n; i++) {

		if (i <= r) z[i] = min(r - i + 1, z[i - l + 1]);

		while (i + z[i] <= n && s[i + z[i]] == s[z[i] + 1]) z[i]++;

		if (i + z[i] - 1 > r) l = i, r = i + z[i] - 1;

	}

	return;

}

```



用 KMP 同样的分析方法，可以得到时间复杂度为 $O(n)$ 。同样可以求 $t$ 与每一个 $s$ 的后缀的 LCP 长度（就是模板题啦）。



```cpp

void exKMP(char *s, int n, char *t, int m) {

	Z(t, m);

	for (int i = 1; i <= n; i++) p[i] = 0;

	for (int i = 1, l = 0, r = 0; i <= n; i++) {

		if (i <= r) p[i] = min(r - i + 1, z[i - l + 1]);

		while (i + p[i] <= n && s[i + p[i]] == t[p[i] + 1]) p[i]++;

		if (i + p[i] - 1 > r) l = i, r = i + p[i] - 1;

	}

	return;

}

```



LCP 也是一种 border 所以可以做字符串匹配。

### [P9576 「TAOI-2」Ciallo～(∠・ω< )⌒★](https://www.luogu.com.cn/problem/P9576)



:::info[题意]

绫地宁宁大战秘密武器！



给定字符串 $s$ ，求有几种至少一个数不同的 $(l,r,l_1,r_1)$ 使得 $1\le l_1\le l\le r\le r_1\le n$ 且 $s(l..l_1)+s(r_1...r)=t$ ，如果 $s(l...r)$ 中 $r<l$ 则其为空串。



$s$ 长为 $n$ 且 $t$ 长为 $m$ 。

:::

:::success[思路+代码]

首先如果 $t$ 是 $s$ 的一个子串那就很容易统计了。



如果 $t$ 被分为了两段，那么我们要使 $LCS(s(l_1...r_1),t)+LCP(s(l_1...r_1))\ge m$ 。



令 $p_1(i)=LCS(s(i...n),t)$ ， $p_2(i)=LCP(s(1...i),t)$ 。这两个可以用 exKMP 求，只需要将字符串倒序正序算即可。



最后我们要统计 $(i,j)$ 的数量，要保证 $j\ge i + m$ 且 $p_1(i)+p_2(j)\ge m$ ，这就是二位数点，用树状数组算即可。



```cpp

#include <bits/stdc++.h>

#define int long long 

using namespace std;

using ll = long long;

const int maxn = 4e5 + 5;

int n, m, z[maxn], p1[maxn], p2[maxn];

char s[maxn], t[maxn];

void Z(char *s, int n) {

	for (int i = 1; i <= n; i++) z[i] = i == 1 ? n : 0;

	for (int i = 2, l = 0, r = 0; i <= n; i++) {

		if (i <= r) z[i] = min(r - i + 1, z[i - l + 1]);

		while (i + z[i] <= n && s[i + z[i]] == s[z[i] + 1]) z[i]++;

		if (i + z[i] - 1 > r) l = i, r = i + z[i] - 1;

	}

	return;

}

void exKMP(char *s, int n, char *t, int m, int *p) {

	Z(t, m);

	for (int i = 1; i <= n; i++) p[i] = 0;

	for (int i = 1, l = 0, r = 0; i <= n; i++) {

		if (i <= r) p[i] = min(r - i + 1, z[i - l + 1]);

		while (i + p[i] <= n && s[i + p[i]] == t[p[i] + 1]) p[i]++;

		if (i + p[i] - 1 > r) l = i, r = i + p[i] - 1;

	}

	return;

}

#define fi first

#define se second

struct BIT {

	ll t[maxn][2];

	void add(int x, ll y) {

		for (int i = x + 1; i <= m + 1; i += i & -i) {

			t[i][0]++; t[i][1] += y;

		}

	}

	pair <ll, ll> query(int x) {

		pair <ll, ll> ans = {0, 0};

		for (int i = x + 1; i; i -= i & -i) {

			ans.fi += t[i][0];

			ans.se += t[i][1];

		}

		return ans;

	} 

} T;

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> (s + 1) >> (t + 1);

	n = strlen(s + 1);

	m = strlen(t + 1);

	exKMP(s, n, t, m, p1);

	reverse(s + 1, s + n + 1);

	reverse(t + 1, t + m + 1);

	exKMP(s, n, t, m, p2);

	reverse(p2 + 1, p2 + n + 1);

	reverse(s + 1, s + n + 1);

	reverse(t + 1, t + m + 1);

	ll ans = 0;

	for (int i = 1; i <= n; i++) {

		if (p2[i] == m) ans += (i - m + 1) * (i - m) / 2 + (n - i + 1) * (n - i) / 2;

	}

	for (int i = m + 1; i <= n; i++) {

		T.add(m - p1[i - m], min(m, p1[i - m] + 1));

		pair <int, int> tmp = T.query(p2[i]);

		ans += tmp.se - max(1ll, m - p2[i]) * tmp.fi;

	}

	cout << ans << '\n';

	return 0;

}

```

# Manacher



Manacher 算法是用来查找回文子串的。首先我们知道，回文串长度有奇有偶，因此我们一般会对字符串进行处理。



```baabcdadcb``` 变成 ```b#a#a#b#c#d#a#d#c#b``` ，这样回文串 ```cdadc``` 变成 ```#c#d#a#d#c``` ，回文串 ```baab``` 变成 ```#b#a#a#b#``` ，我们只用求奇长度回文串即可。



我们将 $d(i)$ 设为以下标 $i$ 为中心的最长回文半径，即 $s(i-d(i)+1...i+d(i)-1)$ 是回文串，现在已知 $d(1...i-1)$ 和右端点最大的回文子区间 $[l,r]$ 该如何求 $d(i)$ 。



- 当 $i>r$ 时，只能暴力扩展了。

- 当 $i\le r$ 时，我们看 $i$ 在 $[l,r]$ 内的对称点 $j=l+r-i$ 。与 $j$ 匹配的回文子串可能超过 $[l,r]$ 的范围，此时 $d(i)$ 的下限即为 $r-i+1$ ，否则 $d(i)$ 下限为 $d(j)$ 即 $d(l+r-1)$ ，由此暴力扩展即可。



要记得随时更新 $[l,r]$ ，代码如下：



```cpp

cin >> (s + 1);

n = strlen(s + 1);

for (int i = 1, l = 0, r = -1; i <= n; i++) {

    int k = i > r ? 1 : min(d[l + r - i], r - i + 1);

    while (i - k >= 1 && i + k <= n && s[i - k] == s[i + k]) k++;

    d[i] = k;

    if (i + k - 1 > r) l = i - k + 1, r = i + k - 1;

}

```



那么从其中提取原字符串的回文子串也很简单，语法题，不多赘述。



$r$ 从 $-1$ 已知不减地增长到 $n$ ，而 $r$ 又是决定 ```while``` 循环次数的，所以此时间复杂度为 $O(n)$ ，外层循环次数为 $O(n)$ ，所以总时间复杂度为线性。

### [P4555 [国家集训队] 最长双回文串](https://www.luogu.com.cn/problem/P4555)



:::info[题意]

求两个相邻回文串长度之和最大。

:::

:::success[思路+代码]

首先进行处理，将 ```baacaabbacabb``` 变成 ```#b#a#a#c#a#a#b#b#a#c#a#b#b#``` ，我们计算 $suc(i)$ 为以 $i$ 开头的最长回文子串长度和 $pre(i)$ 为以 $i$ 结尾的最长回文子串长度，最后枚举每一个 '#' 的位置 $i$ 计算 $\max suc(i)+pre(i)$ 即可。



每次做完 Manacher 算出 $d(i)$ 时，我们将更新 $pre(i+d(i)-1)$ 和 $suc(i-d(i)+1)$ 的值，最后从头到尾处理每一个 '#' 的 $suc(i)=\max(suc(i),suc(i-2)-2)$ ，从尾到头处理每一个 '#' 的 $pre(i)=\max(pre(i),pre(i+2)-2)$ 即可（其实算是动态规划）。



```cpp

#include <bits/stdc++.h>

using namespace std;

const int maxn = 2e5 + 5;

int n, d[maxn], pre[maxn], suc[maxn];

char s[maxn];

int main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> (s + 1);

	n = strlen(s + 1);

	for (int i = n; i >= 1; i--) {

		s[i << 1 | 1] = '#';

		s[i << 1] = s[i];

	} s[1] = '#';

	n = (n << 1 | 1);

	for (int i = 1, l = 0, r = -1; i <= n; i++) {

		int k = i > r ? 1 : min(d[l + r - i], r - i + 1);

		while (i - k >= 1 && i + k <= n && s[i + k] == s[i - k]) k++;

		d[i] = k; 

		if (i + k - 1 > r) l = i - k + 1, r = i + k - 1; 

		pre[i + k - 1] = max(pre[i + k - 1], k - 1);

		suc[i - k + 1] = max(suc[i - k + 1], k - 1);

	}

	for (int i = 1; i <= n; i += 2) suc[i] = max(suc[i], suc[i - 2] - 2);

	for (int i = n; i >= 1; i -= 2) pre[i] = max(pre[i], pre[i + 2] - 2);

	int ans = 0;

	for (int i = 1; i <= n; i += 2) if (pre[i] && suc[i]) ans = max(ans, pre[i] + suc[i]);

	cout << ans << '\n';

	return 0;

} 

```

### [P4287 [SHOI2011] 双倍回文](https://www.luogu.com.cn/problem/P4287)



:::info[题意]

求字符串中由两个相同的偶回文串拼起来的回文子串的最大长度。

:::

:::success[思路+代码]



看似很复杂，但事实上这样的回文串满足以下性质：



- 这个子串是回文串

- 长度为 $4$ 的倍数

- 这个子串的前一半也是回文串



只要发现新区间超过 $[l,r]$ 时向后暴力检查即可。



```cpp

#include <bits/stdc++.h>

using namespace std;

const int maxn = 1e6 + 5;

int n, ans, d[maxn];

char a[maxn];

int main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n >> (a + 1);

	a[0] = '?';

	for (int i = n; i; i--) {

		a[i << 1 | 1] = '#';

		a[i << 1] = a[i];

	}

	a[1] = '#';

	n = n << 1 | 1;

	for (int i = 1, l = 0, r = -1; i <= n; i++) {

		int k = i < r ? min(d[l + r - i], r - i + 1) : 1;

		while (i - k >= 1 && i + k <= n && a[i + k] == a[i - k]) k++;

		d[i] = k;

		if (i + k - 1 > r) {

			if (i % 2) {

				for (int j = max(r, i + 4); j < i + d[i]; j++) {

					if (!((j - i) % 4) && d[i - (j - i) / 2] > (j - i) / 2) {

						ans = max(ans, j - i);

					}

				}

			}

			r = i + k - 1;

			l = i - k + 1;

		}

	}

	cout << ans << '\n';

	return 0; 

}

```

复杂度为什么正确呢？在 ```if (i % 2)``` 中的循环次数总量显然和 ```k``` 的增量差不多，而 ```k``` 的增量是 $O(n)$ 的，所以总复杂度仍然是 $O(n)$ 。

Trie 是一种树状结构（废话），每一个字符串都是树上从根开始的一条链，通过共用前缀维护多个字符串，比如 ```aabaa``` 和 ```aabacd``` 中的 ```aaba``` 在树上就是共用一条链。

其中 ```tr[maxn][26]``` 中的 $26$ 其实指的是字符串可能存在的字符的集合大小，而 ```ID(char c)``` 则是将字符映射为数字的函数，不同题目背景不同处理。

### [P4551 最长异或路径](https://www.luogu.com.cn/problem/P4551)



:::info[题意]

给定一棵树，求异或和最大的路径的异或会。

:::

:::success[思路+代码]

一段路径的异或和连着异或两次会消失，令 $dis(u)$ 为 $1$ 到 $u$ 的异或和，那么点对 $(u,v)$ 之间的异或和为 $dis(u)\otimes dis(v)$ 。



在 01Trie 中插入每一个 $dis(i)$ ，枚举 $i$ 得到其最优答案。



```cpp

#include <bits/stdc++.h>

using namespace std;

const int maxn = 1e5 + 5;

int n, u, v, w, dis[maxn];

vector <pair <int, int>> G[maxn];

#define fi first

#define se second

void dfs(int u, int fa) {

	for (auto g : G[u]) {

		if (g.fi == fa) continue;

		dis[g.fi] = dis[u] ^ g.se;

		dfs(g.fi, u);

	}

	return;

}

struct Trie {

	int tr[maxn << 5][2], idx;

	Trie() {

		memset(tr, 0, sizeof tr);

		idx = 0;

	}

	void insert(int x) {

		int u = 0;

		for (int i = 30; i >= 0; i--) {

			int to = (x >> i) & 1;

			if (!tr[u][to]) tr[u][to] = ++idx;

			u = tr[u][to];

		}

		return;

	}

	int query(int x) {

		int u = 0, ans = 0;

		for (int i = 30; i >= 0; i--) {

			int to = (x >> i) & 1;

			if (tr[u][to ^ 1]) {

				u = tr[u][to ^ 1];

				ans ^= 1 << i;

			} else {

				u = tr[u][to];

			}

		}

		return ans;

	}

} trie;

int main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n;

	for (int i = 1; i < n; i++) {

		cin >> u >> v >> w;

		G[u].push_back({v, w});

		G[v].push_back({u, w});

	}

	dfs(1, 0);

	for (int i = 1; i <= n; i++) {

		trie.insert(dis[i]);

	}

	int ans = 0;

	for (int i = 1; i <= n; i++) {

		ans = max(ans, trie.query(dis[i]));

	}

	cout << ans << '\n';

	return 0; 

}

```

## fail 树



对于 KMP 求出的 border 数组，将 $i$ 与 $border_i$ 连边，得到一个以 $0$ 为根节点的 border 树，即 fail 树。在 fail 树上，点 $u$ 的祖先都是它的 border 长度。



## KMP 自动机



KMP 自动机是弱化版的 AC 自动机，相当于单串的 AC 自动机。先建一个 fail 树，设 $nxt_{i,j}$ 表示从 $i$ 开始沿着 fail 树向上跳得到的最大的位置 $k$ 使 $s_{k+1}=j$ 。



注意到此时我们可以把 KMP 求的过程表示为：对于 $i$ ，从 $fail_{i−1}$ 开始沿着 fail 树向上跳找到的最大的位置 $k$ 使 $s_{k+1}=s_i$ ，若不存在则为 $0$ 。这也可以让我们得到 $fail_i=nxt_{fail_{i−1,s_i}}$ 。
