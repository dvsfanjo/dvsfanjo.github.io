## 思路



这是今天刚做到的 NOIP 模拟赛题，先看前面两个 Subtask 启示我们什么。



在 Subtask 1 中 $n\le1000$ 可以直接暴力加边，每次询问则暴力跑 dijkstra 得到答案。



在 Subtask 2 中每个询问的 $a$ 固定，则说明最短路从多源变为单源，每个结点是在两个原有结点的基础上添加的，所以到达新结点的最短路只能由两个原有结点处松弛而来。



这两个 Subtask 分别启示我们：跑最短路的做法是没有前途的，但是每个结点的最短路都由原先的两个结点松弛而来；这说明每次想要高效询问，可以把最短路转化为树上距离问题，而建树则要用到新结点来源的两个老结点。



最开始图上只有边 $(1,2)$，将其视作辅助树的根节点，每次给定新点 $p$ 和原图中的边 $(a,b)$，就将 $(a,b)$ 在辅助树上对应的结点下，挂上上个子节点，分别代表原图中的 $(a,p)$ 和 $(b,p)$，这样就可以将原图中的边映射为辅助树上的结点。



原图如下时：



```

1------2

 \    /

  \  /

    3

```



辅助树如下：



```

    (1,2)

    /   \

   /     \

(1,3)   (2,3)

```



转化为树后，发现辅助树上的结点都代表一条边和两个原图上的结点，于是为了方便维护原图上的最短路，将辅助树上的边权要维护辅助树上相邻结点代表的原图上的 $4$ 个结点直接的距离，比如 $(1,2)$ 与 $(1,3)$ 直接的边权可以这样表示：



$$

\begin{bmatrix} 

dis(p_1,p_1) & dis(p_1,p_3) \\

dis(p_2,p_1) & dis(p_2,p_3) 

\end{bmatrix}

$$



若反过来从 $(1,3)$ 到 $(1,2)$ 的边权，则交换原边权从左下到右上对角线上的两个元素即可，故从 $(2,3)$ 到 $(1,2)$ 的边权为：



$$

\begin{bmatrix} 

dis(p_2,p_1) & dis(p_2,p_2) \\

dis(p_3,p_1) & dis(p_3,p_2) 

\end{bmatrix}

$$



考虑如何合并这个边权，用 $(2,3)\to(1,2)$ 和 $(1,2)\to(1,3)$ 的边权合并，应该得到：



$$

\begin{bmatrix} 

\min(dis_{2,1}+dis_{1,1},dis_{2,2}+dis_{2,1}) & \min(dis_{2,1}+dis_{1,3},dis_{1,3}+dis_{2,3}) \\

\min(dis_{3,1}+dis_{1,1},dis_{3,2}+dis_{2,1}) & 

\min(dis_{3,1}+dis_{1,3},dis_{3,2}+dis_{2,3})

\end{bmatrix}

$$



惊奇的发现，这就是两个矩阵的 min-plus 运算且可以推广成不同情况，这是有结合律的，于是通过倍增求 LCA 的套路维护链上的矩阵运算的结果，交换左下右上对角线可以交换矩阵运算的整体方向，且运算后表示的含义不变。



具体实现时，要分别记录每个结点第一次被新建的时候，属于哪两条边，用这两条边对应在辅助树中的结点跑 LCA 即可，而且还有可能查询的结点不需要同时跳到 LCA ，而是最后同时在兄弟结点，这也是一种情况。



具体实现可以看细节，矩阵运算我为了减少常数直接定义了 $4$ 个变量。



## 代码



```cpp

#include <bits/stdc++.h>

#define int long long

#define fi first

#define se second

using namespace std;

const int maxn = 2e5 + 5;

const double inf = 1e14;

int n = 2, m = 1, q, dep[maxn], f[maxn][20], born[maxn];

pair <int, int> edge[maxn];

vector <int> G[maxn];

unordered_map <int, unordered_map <int, int>> mp;

struct point { 

	double x, y;

	inline void read() { cin >> x >> y; }

	point(double _x = 0.0, double _y = 0.0) { x = _x, y = _y; }

} p[maxn];

inline double calc(point a, point b) {

	double dx = a.x - b.x, dy = a.y - b.y;

	return sqrt(dx * dx + dy * dy);

}

struct query {

	char op; double x, y; int a, b;

	inline void read() {

		cin >> op;

		if (op == 'd') {

			cin >> x >> y >> a >> b;

			p[++n] = point(x, y);

			if (a > b) swap(a, b);

			int v = mp[a][b];

			edge[++m] = make_pair(a, n), mp[a][n] = born[n] = m;

			edge[++m] = make_pair(b, n), mp[b][n] = m;

			G[v].push_back(m - 1), G[v].push_back(m);

		} else {

			cin >> a >> b;

			if (a > b) swap(a, b);

		}

		return;

	}

} qr[maxn];

struct Mat {

	double val00, val01, val10, val11;

	Mat(double _val00 = 0.0, double _val01 = inf, 

		double _val10 = inf, double _val11 = 0.0) {

		val00 = _val00, val01 = _val01, 

		val10 = _val10, val11 = _val11;

	}

	friend Mat operator * (Mat a, Mat b) {

		return Mat(min(a.val00 + b.val00, a.val01 + b.val10), 

				   min(a.val00 + b.val01, a.val01 + b.val11), 

				   min(a.val10 + b.val00, a.val11 + b.val10), 

				   min(a.val10 + b.val01, a.val11 + b.val11));

	}

} g[maxn][20];

inline Mat make(pair <int, int> u, pair <int, int> v) {

	return Mat(calc(p[u.fi], p[v.fi]), calc(p[u.fi], p[v.se]), 

			   calc(p[u.se], p[v.fi]), calc(p[u.se], p[v.se]));

}

inline Mat rev(Mat a) {

	swap(a.val01, a.val10);

	return a;

}

void dfs(int u, int fa) {

	dep[u] = dep[fa] + 1, f[u][0] = fa;

	if (fa) g[u][0] = make(edge[u], edge[fa]);

	for (int i = 1; i < 20; i++) {

		f[u][i] = f[f[u][i - 1]][i - 1];

		g[u][i] = g[u][i - 1] * g[f[u][i - 1]][i - 1];

	}

	for (int v : G[u]) dfs(v, u);

	return;

}

double solve(int a, int b, double ans = inf) {

	if (a == b) return 0.0;

	if (mp[a][b]) return calc(p[a], p[b]); 

	vector <int> Aedge, Bedge;

	if (a == 1 || a == 2) Aedge.push_back(1);

	else Aedge.push_back(born[a]), Aedge.push_back(born[a] + 1); 

	if (b == 1 || b == 2) Bedge.push_back(1);

	else Bedge.push_back(born[b]), Bedge.push_back(born[b] + 1); 

	auto get_val = [&](Mat &a, int x, int y)->double{

		if (x == 0 && y == 0) return a.val00;

		if (x == 0 && y == 1) return a.val01;

		if (x == 1 && y == 0) return a.val10;

		if (x == 1 && y == 1) return a.val11;

	};

	auto get = [&](int u, int v, int a, int b)->double{

		if (dep[u] < dep[v]) swap(u, v), swap(a, b);

		bool x = edge[u].se == a, y = edge[v].se == b;

		Mat A, B;

		for (int i = 19; i >= 0; i--) {

			if (dep[f[u][i]] >= dep[v]) {

				A = A * g[u][i];

				u = f[u][i];

			}

		}

		if (u == v) return get_val(A, x, y);

		for (int i = 19; i >= 0; i--) {

			if (f[u][i] != f[v][i]) {

				A = A * g[u][i], B = B * g[v][i], 

				u = f[u][i], v = f[v][i];

			}

		}

		double ans = inf;

		if ((u >> 1) == (v >> 1)) {

			ans = min(ans, get_val(A, x, 1) + get_val(B, 1, 1));

		}

		A = A * g[u][0], B = B * g[v][0];

		A = A * rev(B);

		ans = min(ans, get_val(A, x, y));

		return ans;

	};

	for (int u : Aedge) for (int v : Bedge) {

		ans = min(ans, get(u, v, a, b));

	}

	return ans;

} 

signed main() { 

	ios::sync_with_stdio(0);

	cin.tie(0), cout.tie(0);

	p[1].read(), p[2].read(),

	edge[1] = make_pair(1, 2),  

	mp[1][2] = 1, cin >> q, 

	born[1] = born[2] = 1; 

	for (int i = 1; i <= q; i++) {

		qr[i].read();

	}

	dfs(1, 0);

	for (int i = 1; i <= q; i++) {

		if (qr[i].op == 'd') continue;

		cout << fixed << setprecision(6) << solve(qr[i].a, qr[i].b) << '\n'; 

	}

	return 0;

}
```
