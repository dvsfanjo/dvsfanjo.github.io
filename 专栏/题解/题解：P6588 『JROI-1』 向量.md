中考前最后一篇题解



## 思路



注意到两个询问的求和都有先后顺序，使用类似 CDQ 分治的思想，求解一段区间的答案，将其分为两份，加上左一份对右一份的贡献。



事实上贡献只是左右区间向量之和的点乘、叉乘，因为这俩都满足结合律，但是还是推一下式子。



将区间 $i$ 的点乘之和记为 $d_i$ ，叉乘之和记为 $c_i$ ，横坐标之和为 $sx_i$ ，纵坐标之和为 $sy_i$ 。



### 点乘



结论如下：



$$d_i=d_{ls}+d_{rs}+sx_{ls}sx_{rs}+sy_{ls}sy_{rs}$$



:::info[证明]

贡献为：



$$\sum_{i\in ls}\sum_{j\in rs}(x_ix_j+y_iy_j)$$



即



$$\sum_{i\in ls}\sum_{j\in rs}x_ix_j+\sum_{i\in ls}\sum_{j\in rs}y_iy_j$$



换位



$$\sum_{i\in ls}x_i\sum_{j\in rs}x_j+\sum_{i\in ls}y_i\sum_{j\in rs}y_j$$



即



$$sx_{ls}sx_{rs}+sy_{ls}sy_{rs}$$

:::



### 叉乘



结论如下：



$$c_i=c_{ls}+c_{rs}+sx_{ls}sy_{rs}-sy_{ls}sx_{rs}$$



:::info[证明]

贡献为：



$$\sum_{i\in ls}\sum_{j\in rs}(x_iy_j-y_ix_j)$$



即



$$\sum_{i\in ls}\sum_{j\in rs}x_iy_j-\sum_{i\in ls}\sum_{j\in rs}y_ix_j$$



换位



$$\sum_{i\in ls}x_i\sum_{j\in rs}y_j-\sum_{i\in ls}y_i\sum_{j\in rs}x_j$$



即



$$sx_{ls}sy_{rs}-sy_{ls}sx_{rs}$$

:::



## 实现



显然这两个可以直接用线段树维护，其他简单的单点操作随便做就行。



```cpp

#include <bits/stdc++.h>

#define int long long 

using namespace std;

const int maxn = 1e5 + 5;

int n, m, op, pos, x, y, lamda, l, r;

#define ls (i << 1)

#define rs (ls | 1)

#define mid (l + r >> 1)

struct vec {

	int x, y;

	vec(int a = 0, int b = 0) { x = a, y = b;}

	friend vec operator + (vec a, vec b) {

		return vec(a.x + b.x, a.y + b.y);

	}

	friend vec operator - (vec a, vec b) {

		return vec(a.x - b.x, a.y - b.y);

	}

	friend vec operator * (int a, vec b) {

		return vec(a * b.x, a * b.y); 

	}

};

inline int dot(vec a, vec b) {return a.x * b.x + a.y * b.y;}

inline int cro(vec a, vec b) {return a.x * b.y - a.y * b.x;}

struct node {

	int d, c;

	vec a;

} t[maxn << 2];

inline node merge(node x, node y) {

	node ret;

	ret.a = x.a + y.a;

	ret.d = x.d + y.d + dot(x.a, y.a);

	ret.c = x.c + y.c + cro(x.a, y.a);

	return ret;

} 

void pushup(int i) {

	t[i] = merge(t[ls], t[rs]);

	return;

}

void build(int l, int r, int i) {

	if (l == r) {

		cin >> t[i].a.x >> t[i].a.y;

		t[i].d = t[i].c = 0;

	} else {

		build(l, mid, ls);

		build(mid + 1, r, rs);

		pushup(i);

	}

	return;

}

void update(int l, int r, int i, int pos, int op) {

	if (l == r) {

		if (op == 1) t[i].a = t[i].a + vec(x, y);

		if (op == 2) t[i].a = t[i].a - vec(x, y);

		if (op == 3) t[i].a = lamda * t[i].a;

	} else {

		if (pos <= mid) update(l, mid, ls, pos, op);

		else update(mid + 1, r, rs, pos, op);

		pushup(i);

	}

	return;

}

node query(int l, int r, int i, int a, int b) {

	if (a <= l && r <= b) {

		return t[i];

	} else {

		if (b <= mid) {

			return query(l, mid, ls, a, b);

		} else if (mid < a) {

			return query(mid + 1, r, rs, a, b);

		} else {

			return merge(query(l, mid, ls, a, b), query(mid + 1, r, rs, a, b));

		}

	}

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n >> m;

	build(1, n, 1);

	while (m--) {

		cin >> op;

		if (op == 1 || op == 2) {

			cin >> pos >> x >> y;

			update(1, n, 1, pos, op);

		}

		if (op == 3) {

			cin >> pos >> lamda;

			update(1, n, 1, pos, op);

		}

		if (op == 4 || op == 5) {

			cin >> l >> r;

			node ans = query(1, n, 1, l, r);

			if (op == 4) cout << ans.d << '\n';

			if (op == 5) cout << ans.c << '\n';

		}

	}

	return 0;

}
```
