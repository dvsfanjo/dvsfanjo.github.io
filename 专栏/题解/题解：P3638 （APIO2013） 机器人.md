## 思路



充分联想，竟然机器人要连续编号才能变成复合机器人，容易联想到和区间动态规划有关。



设 $f(l,r,i,j)$ 表示在坐标 $(i,j)$ 上得到 $[l,r]$ 的复合机器人需要的最小步数。



得到 $f(l,r,i,j)$ 有两种转移方式：



1. 由两个 $f(l,k,i,j)$ 和 $f(k+1,r,i,j)$ 得到，最小步数则是 $\min_{k\in [l,r)}(f(l,k,i,j)+f(k+1,r,i,j))$ 。

2. 由一个 $f(l,r,x,y)$ 通过移动得到。我们要预处理每个位置向每个方向移动的终点，使用记忆化搜索完成。我们通过将所有位置的 $[l,r]$ 复合机器人合并后，使用改进的 bfs 来更新。



关于改进的 bfs 有以下思考：



首先我想从 $f(l,r,x,y)$ 最小的 $(x,y)$ 为起点，做 bfs 来更新所有的点，但是一旦在某个点无法松弛，后面的点都无法得到更新。



于是又想，可以对每一个点都做一遍 bfs 来更新，我们广搜前，可以将所有点根据其 $f(l,r,x,y)$ 从小到大排序并放在另一个队列中，广搜时，如果当前更新到的点值比另一个队列队首的点值大，我们优先从点值较小的点进行进一步的更新。



补充几点细节喵：



1. 无解输出 -1 喵。

2. 每个 $f(l,r,x,y)$ 的初始值赋为正无穷喵。

3. 吸氧喵。



做完啦喵！



## 代码



```cpp

#include <bits/stdc++.h>

using namespace std;

using pii = pair <int, int>;

#define fi first

#define se second

const int inf = 1e9;

int n, h, w, f[10][10][505][505], vis[505][505][4], viss[505][505];

int dx[4] = {1, 0, -1, 0}, dy[4] = {0, 1, 0, -1};

char s[505][505];

pii to[505][505][4];

pii dfs(int x, int y, int dir) {

	if (vis[x][y][dir]) return to[x][y][dir];

	vis[x][y][dir] = true;

	int xx = x + dx[dir], yy = y + dy[dir];

	if (xx < 1 || xx > h || yy < 1 || yy > w) return to[x][y][dir] = {x, y};

	if (s[xx][yy] == 'x') return to[x][y][dir] = {x, y};

	if (s[xx][yy] == 'A') return to[x][y][dir] = dfs(xx, yy, (dir + 1) & 3);

	if (s[xx][yy] == 'C') return to[x][y][dir] = dfs(xx, yy, (dir + 3) & 3);

	return to[x][y][dir] = dfs(xx, yy, dir);

}

struct node {

	pii a; int b;

	friend bool operator < (node x, node y) {

		return x.b < y.b;

	}

};

void bfs(int l, int r) {

	queue <node> q1, q2;

	vector <node> vec;

	for (int i = 1; i <= h; i++) {

		for (int j = 1; j <= w; j++) {

			vec.push_back({{i, j}, f[l][r][i][j]});

		}

	}

	sort(vec.begin(), vec.end());

	for (auto v : vec) q1.push(v);

	while (q1.size() || q2.size()) {

		pii u; int v;

		if (q1.empty()) u = q2.front().a, v = q2.front().b, q2.pop();

        else if (q2.empty()) u = q1.front().a, v = q1.front().b, q1.pop();

        else if (q1.front().b < q2.front().b) u = q1.front().a, v = q1.front().b, q1.pop();

        else u = q2.front().a, v = q2.front().b, q2.pop();

        if (viss[u.fi][u.se]) continue;

        viss[u.fi][u.se] = true;

        for (int d = 0; d < 4; d++) {

        	int xx = to[u.fi][u.se][d].fi, yy = to[u.fi][u.se][d].se;

        	if (f[l][r][xx][yy] > v + 1) {

        		f[l][r][xx][yy] = v + 1;

        		q2.push({{xx, yy}, f[l][r][xx][yy]});

			}

		}

	}

	memset(viss, 0, sizeof viss);

	return;

}

int main() {

	ios::sync_with_stdio(0);

	cin.tie(0); cout.tie(0);

	cin >> n >> w >> h;

	for (int i = 1; i <= h; i++) {

		cin >> (s[i] + 1);

	}

	for (int i = 1; i <= h; i++) {

		for (int j = 1; j <= w; j++) {

			for (int d = 0; d < 4; d++) {

				dfs(i, j, d);

			}

			for (int k = 1; k <= n; k++) {

				if (s[i][j] - '0' == k) f[k][k][i][j] = 0;

				else f[k][k][i][j] = inf;

			}

		}

	}

	for (int i = 1; i <= n; i++) bfs(i, i);

	for (int len = 2; len <= n; len++) {

		for (int l = 1, r = len; r <= n; l++, r++) {

			for (int i = 1; i <= h; i++) {

				for (int j = 1; j <= w; j++) {

					f[l][r][i][j] = inf;

					for (int k = l; k < r; k++) {

						f[l][r][i][j] = min(f[l][r][i][j], f[l][k][i][j] + f[k + 1][r][i][j]);

					}

				}

			}

			bfs(l, r);

		}

	}

	int ans = inf;

	for (int i = 1; i <= h; i++) {

		for (int j = 1; j <= w; j++) {

			ans = min(ans, f[1][n][i][j]);

		} 

	}

	cout << (ans == inf ? -1 : ans) << '\n';

	return 0;

}
```
