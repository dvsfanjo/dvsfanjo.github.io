## 思路



考察了分解质因数的知识点。首先考虑如何分解质因数，只要线性筛一遍，将值域内所有数的最小素因数得出，再一个个除得到分解质因数的结果，因为每个素数大小至少为 $2$ 则试除法最多进行 $O(\log n)$ 次。将所有质因数放入桶中比对即可。



## 代码



```cpp

#include <bits/stdc++.h>

#define int long long

using namespace std;

const int maxn = 1e7 + 5;

int T, x, y, f[maxn], p[maxn];

vector <int> prime;

vector <int> solve(int x) {

	vector <int> ans;

	while (x != 1) {

		int tmp = p[x];

		while (x % tmp == 0) x /= tmp;

		ans.push_back(tmp);

	}

	return ans;

}

signed main() {

	ios::sync_with_stdio(0);

	cin.tie(0);

	cin >> T;

	for (int i = 2; i <= 10000000; i++) {

		if (!f[i]) {

			prime.push_back(i);

			p[i] = i;

		} 

		for (auto P : prime) {

			if (i * P > 10000000) break;

			f[i * P] = 1;

			p[i * P] = P;

			if (i % P == 0) break;

		}

	}

	unordered_map <int, bool> mp1, mp2;

	while (T--) {

		cin >> x >> y;

		vector <int> c1 = solve(x), c2 = solve(y);

		mp1.clear(), mp2.clear();

		int ans = 1;

		for (auto tmp : c1) mp1[tmp] = 1;

		for (auto tmp : c2) mp2[tmp] = 1;

		for (auto tmp : c1) if (!mp2[tmp]) ans *= tmp;

		for (auto tmp : c2) if (!mp1[tmp]) ans *= tmp;

		cout << ans << '\n';

	}

	return 0;

}
```
